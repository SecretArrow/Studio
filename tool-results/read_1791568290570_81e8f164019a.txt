    79→const CanvasEditor = forwardRef<EditorHandle, EditorProps>(function CanvasEditor(
    80→  { project, initialDoc, role, onDocChange, registerHandle },
    81→  _ref,
    82→) {
    83→  const canEdit = role === "owner" || role === "editor"
    84→  const [doc, setDoc] = useState<DesignDoc>(initialDoc)
    85→  const [pageIndex, setPageIndexState] = useState(0)
    86→  const [selectedIds, setSelectedIds] = useState<string[]>([])
    87→  const [view, setViewState] = useState<ViewState>({ ...DEFAULT_VIEW })
    88→  const [historyState, setHistoryState] = useState({ canUndo: false, canRedo: false })
    89→  const [activeTab, setActiveTab] = useState<TabId | null>(null)
    90→  const [editingId, setEditingId] = useState<string | null>(null)
    91→  const [contextMenu, setContextMenu] = useState<{ x: number; y: number; elementId: string | null } | null>(null)
    92→  const [viewport, setViewport] = useState({ w: 0, h: 0 })
    93→  const [propsSheetOpen, setPropsSheetOpen] = useState(false)
    94→
    95→  const docRef = useRef<DesignDoc>(initialDoc)
    96→  const pageIndexRef = useRef(0)
    97→  const viewRef = useRef(view)
    98→  useEffect(() => {
    99→    viewRef.current = view
   100→  }, [view])
   101→  const [history] = useState(() => new HistoryStore(initialDoc, 60))
   102→  const clipboard = useRef<Clipboard | null>(null)
   103→  const dirtyRef = useRef(false)
   104→  const firstFitDone = useRef(false)
   105→
   106→  const page: PageModel = doc.pages[Math.min(pageIndex, doc.pages.length - 1)] ?? doc.pages[0]
   107→  const selection = useMemo(() => page.elements.filter((e) => selectedIds.includes(e.id)), [page, selectedIds])
   108→
   109→  const syncHistoryState = useCallback(() => {
   110→    setHistoryState({ canUndo: history.canUndo, canRedo: history.canRedo })
   111→  }, [history])
   112→
   113→  /* ------------------------- core mutations ------------------------- */
   114→
   115→  const commit = useCallback(
   116→    (next: DesignDoc, coalesceKey?: string) => {
   117→      docRef.current = next
   118→      dirtyRef.current = true
   119→      setDoc(next)
   120→      history.push(next, coalesceKey)
   121→      syncHistoryState()
   122→      onDocChange(next)
   123→    },
   124→    [history, onDocChange, syncHistoryState],
   125→  )
   126→
   127→  const mutatePage = useCallback(
   128→    (fn: (page: PageModel) => PageModel, coalesceKey?: string) => {
   129→      const prev = docRef.current
   130→      const next: DesignDoc = {
   131→        ...prev,
   132→        pages: prev.pages.map((p, i) => (i === pageIndexRef.current ? fn(p) : p)),
   133→      }
   134→      commit(next, coalesceKey)
   135→    },
   136→    [commit],
   137→  )
   138→
   139→  const ensureTextHeight = (el: DesignElement): DesignElement => {
   140→    if (el.type !== "text") return el
   141→    const t = el as TextElement
   142→    const h = measureTextBlockHeight(
   143→      {
   144→        text: t.text,
   145→        fontFamily: t.fontFamily,
   146→        fontSize: t.fontSize,
   147→        fontWeight: t.fontWeight,
   148→        italic: t.italic,
   149→        uppercase: t.uppercase,
   150→        lineHeight: t.lineHeight,
   151→        letterSpacing: t.letterSpacing,
   152→        listStyle: t.listStyle,
   153→      },
   154→      t.width,
   155→    )
   156→    return { ...t, height: Math.max(t.height, Math.round(h)) }
   157→  }
   158→
   159→  const updateElements = useCallback(
   160→    (patches: ElementPatch[], opts?: { coalesceKey?: string }) => {
   161→      if (!canEdit || patches.length === 0) return
   162→      const map = new Map(patches.map((p) => [p.id, p.patch]))
   163→      mutatePage(
   164→        (p) => ({
   165→          ...p,
   166→          elements: p.elements.map((el) => {
   167→            const patch = map.get(el.id)
   168→            if (!patch) return el
   169→            const merged = { ...el, ...patch } as DesignElement
   170→            const textKeys = ["text", "fontSize", "fontFamily", "fontWeight", "italic", "uppercase", "lineHeight", "letterSpacing", "listStyle", "width"]
   171→            if (merged.type === "text" && textKeys.some((k) => k in patch)) return ensureTextHeight(merged)
   172→            return merged
   173→          }),
   174→        }),
   175→        opts?.coalesceKey,
   176→      )
   177→    },
   178→    [canEdit, mutatePage],
   179→  )
   180→
   181→  const dropPos = useCallback(
   182→    (w: number, h: number) => {
   183→      const v = viewRef.current
   184→      const vp = viewport.w > 0 ? viewport : { w: 900, h: 600 }
   185→      const cx = (vp.w / 2 - v.panX) / v.zoom
   186→      const cy = (vp.h / 2 - v.panY) / v.zoom
   187→      return {
   188→        x: Math.max(12, Math.round(cx - w / 2)),
   189→        y: Math.max(12, Math.round(cy - h / 2)),
   190→      }
   191→    },
   192→    [viewport],
   193→  )
   194→
   195→  const addElements = useCallback(
   196→    (els: DesignElement[], select = true) => {
   197→      if (!canEdit || els.length === 0) return
   198→      const prepared = els.map(ensureTextHeight)
   199→      mutatePage((p) => ({ ...p, elements: [...p.elements, ...prepared] }))
   200→      if (select) setSelectedIds(prepared.map((el) => el.id))
   201→    },
   202→    [canEdit, mutatePage],
   203→  )
   204→
   205→  const deleteElements = useCallback(
   206→    (ids: string[]) => {
   207→      if (!canEdit || ids.length === 0) return
   208→      const set = new Set(ids)
   209→      mutatePage((p) => ({ ...p, elements: p.elements.filter((e) => !set.has(e.id)) }))
   210→      setSelectedIds((prev) => prev.filter((id) => !set.has(id)))
   211→      setEditingId(null)
   212→    },
   213→    [canEdit, mutatePage],
   214→  )
   215→
   216→  const duplicateElements = useCallback(
   217→    (ids: string[]) => {
   218→      if (!canEdit || ids.length === 0) return
   219→      const set = new Set(ids)
   220→      const source = page.elements.filter((e) => set.has(e.id))
   221→      if (source.length === 0) return
   222→      const gidMap = new Map<string, string>()
   223→      const clones = source.map((el) => {
   224→        let groupId = el.groupId
   225→        if (groupId) {
   226→          if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   227→          groupId = gidMap.get(groupId)
   228→        }
   229→        return { ...el, id: uid("dup"), x: el.x + 24, y: el.y + 24, groupId } as DesignElement
   230→      })
   231→      mutatePage((p) => ({ ...p, elements: [...p.elements, ...clones] }))
   232→      setSelectedIds(clones.map((c) => c.id))
   233→    },
   234→    [canEdit, mutatePage, page.elements],
   235→  )
   236→
   237→  const selectIds = useCallback((ids: string[], additive = false) => {
   238→    setEditingId(null)
   239→    setSelectedIds((prev) => {
   240→      if (!additive) return ids
   241→      const set = new Set(prev)
   242→      for (const id of ids) {
   243→        if (set.has(id)) set.delete(id)
   244→        else set.add(id)
   245→      }
   246→      return Array.from(set)
   247→    })
   248→  }, [])
   249→
   250→  const reorder = useCallback(
   251→    (ids: string[], dir: "front" | "forward" | "backward" | "back") => {
   252→      if (!canEdit || ids.length === 0) return
   253→      const set = new Set(ids)
   254→      mutatePage((p) => {
   255→        let elements = [...p.elements]
   256→        if (dir === "front") {
   257→          const picked = elements.filter((e) => set.has(e.id))
   258→          elements = [...elements.filter((e) => !set.has(e.id)), ...picked]
   259→        } else if (dir === "back") {
   260→          const picked = elements.filter((e) => set.has(e.id))
   261→          elements = [...picked, ...elements.filter((e) => !set.has(e.id))]
   262→        } else if (dir === "forward") {
   263→          for (let i = elements.length - 2; i >= 0; i -= 1) {
   264→            if (set.has(elements[i].id) && !set.has(elements[i + 1].id)) {
   265→              ;[elements[i], elements[i + 1]] = [elements[i + 1], elements[i]]
   266→            }
   267→          }
   268→        } else {
   269→          for (let i = 1; i < elements.length; i += 1) {
   270→            if (set.has(elements[i].id) && !set.has(elements[i - 1].id)) {
   271→              ;[elements[i], elements[i - 1]] = [elements[i - 1], elements[i]]
   272→            }
   273→          }
   274→        }
   275→        return { ...p, elements }
   276→      })
   277→    },
   278→    [canEdit, mutatePage],
   279→  )
   280→
   281→  const groupSelected = useCallback(() => {
   282→    if (!canEdit || selection.length < 2) return
   283→    const gid = uid("grp")
   284→    updateElements(selection.map((e) => ({ id: e.id, patch: { groupId: gid } })))
   285→  }, [canEdit, selection, updateElements])
   286→
   287→  const ungroupSelected = useCallback(() => {
   288→    if (!canEdit || selection.length === 0) return
   289→    updateElements(selection.map((e) => ({ id: e.id, patch: { groupId: undefined } })))
   290→  }, [canEdit, selection, updateElements])
   291→
   292→  const renameElement = useCallback(
   293→    (id: string, name: string) => {
   294→      updateElements([{ id, patch: { name: name.trim() || undefined } }])
   295→    },
   296→    [updateElements],
   297→  )
   298→
   299→  /* ------------------------- clipboard ------------------------- */
   300→
   301→  const copySelection = useCallback(
   302→    (cut = false) => {
   303→      if (selection.length === 0) return
   304→      clipboard.current = { elements: JSON.parse(JSON.stringify(selection)) as DesignElement[] }
   305→      if (cut) deleteElements(selectedIds)
   306→    },
   307→    [deleteElements, selection, selectedIds],
   308→  )
   309→
   310→  const pasteClipboard = useCallback(() => {
   311→    const clip = clipboard.current
   312→    if (!canEdit || !clip || clip.elements.length === 0) return
   313→    const gidMap = new Map<string, string>()
   314→    const clones = clip.elements.map((el) => {
   315→      let groupId = el.groupId
   316→      if (groupId) {
   317→        if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   318→        groupId = gidMap.get(groupId)
   319→      }
   320→      return { ...el, id: uid("paste"), x: el.x + 24, y: el.y + 24, groupId } as DesignElement
   321→    })
   322→    addElements(clones)
   323→  }, [addElements, canEdit])
   324→
   325→  /* ------------------------- pages ------------------------- */
   326→
   327→  const setPageIndex = useCallback((i: number) => {
   328→    const docCurrent = docRef.current
   329→    const clamped = Math.max(0, Math.min(i, docCurrent.pages.length - 1))
   330→    pageIndexRef.current = clamped
   331→    setPageIndexState(clamped)
   332→    setSelectedIds([])
   333→    setEditingId(null)
   334→  }, [])
   335→
   336→  const addPage = useCallback(
   337→    (duplicate: boolean) => {
   338→      if (!canEdit) return
   339→      const prev = docRef.current
   340→      const current = prev.pages[pageIndexRef.current]
   341→      let newPage: PageModel
   342→      if (duplicate) {
   343→        const gidMap = new Map<string, string>()
   344→        newPage = {
   345→          ...current,
   346→          id: uid("page"),
   347→          name: `${current.name} copy`,
   348→          elements: current.elements.map((el) => {
   349→            let groupId = el.groupId
   350→            if (groupId) {
   351→              if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   352→              groupId = gidMap.get(groupId)
   353→            }
   354→            return { ...el, id: uid("el"), groupId } as DesignElement
   355→          }),
   356→        }
   357→      } else {
   358→        newPage = createPage({ name: `Page ${prev.pages.length + 1}`, background: current.background })
   359→      }
   360→      const pages = [...prev.pages]
   361→      pages.splice(pageIndexRef.current + 1, 0, newPage)
   362→      commit({ ...prev, pages })
   363→      setPageIndex(pageIndexRef.current + 1)
   364→    },
   365→    [canEdit, commit, setPageIndex],
   366→  )
   367→
   368→  const deletePage = useCallback(
   369→    (index: number) => {
   370→      if (!canEdit) return
   371→      const prev = docRef.current
   372→      if (prev.pages.length <= 1) return
   373→      const pages = prev.pages.filter((_, i) => i !== index)
   374→      commit({ ...prev, pages })
   375→      setPageIndex(Math.max(0, Math.min(pageIndexRef.current > index ? pageIndexRef.current - 1 : pageIndexRef.current, pages.length - 1)))
   376→    },
   377→    [canEdit, commit, setPageIndex],
   378→  )
   379→
   380→  const movePage = useCallback(
   381→    (from: number, to: number) => {
   382→      if (!canEdit || from === to) return
   383→      const prev = docRef.current
   384→      const pages = [...prev.pages]
   385→      const [moved] = pages.splice(from, 1)
   386→      pages.splice(to, 0, moved)
   387→      commit({ ...prev, pages })
   388→      if (pageIndexRef.current === from) setPageIndex(to)
   389→      else setPageIndex(pageIndexRef.current)
   390→    },
   391→    [canEdit, commit, setPageIndex],
   392→  )
   393→
   394→  const renamePage = useCallback(
   395→    (index: number, name: string) => {
   396→      if (!canEdit) return
   397→      const prev = docRef.current
   398→      commit({
   399→        ...prev,
   400→        pages: prev.pages.map((p, i) => (i === index ? { ...p, name } : p)),
   401→      })
   402→    },
   403→    [canEdit, commit],
   404→  )
   405→
   406→  const setPageBackground = useCallback(
   407→    (bg: BackgroundSpec) => {
   408→      if (!canEdit) return
   409→      mutatePage((p) => ({ ...p, background: bg }))
   410→    },
   411→    [canEdit, mutatePage],
   412→  )
   413→
   414→  /* ------------------------- view ------------------------- */
   415→
   416→  const setView = useCallback((patch: Partial<ViewState>) => {
   417→    setViewState((prev) => ({ ...prev, ...patch }))
   418→  }, [])
   419→
   420→  const onViewportResize = useCallback((w: number, h: number) => {
   421→    setViewport((prev) => (prev.w === w && prev.h === h ? prev : { w, h }))
   422→  }, [])
   423→
   424→  const fitToScreen = useCallback(() => {
   425→    if (viewport.w <= 0 || viewport.h <= 0) return
   426→    const d = docRef.current
   427→    const pad = 56
   428→    const zoom = clampZoom(Math.min((viewport.w - pad) / d.width, (viewport.h - pad) / d.height))
   429→    setViewState((prev) => ({
   430→      ...prev,
   431→      zoom,
   432→      panX: (viewport.w - d.width * zoom) / 2,
   433→      panY: (viewport.h - d.height * zoom) / 2,
   434→    }))
   435→  }, [viewport])
   436→
   437→  useEffect(() => {
   438→    if (!firstFitDone.current && viewport.w > 0) {
   439→      firstFitDone.current = true
   440→      fitToScreen()
   441→    }
   442→  }, [viewport, fitToScreen])
   443→
   444→  const setZoom = useCallback(
   445→    (z: number) => {
   446→      const target = clampZoom(z)
   447→      const v = viewRef.current
   448→      const vp = viewport.w > 0 ? viewport : { w: 900, h: 600 }
   449→      const cx = vp.w / 2
   450→      const cy = vp.h / 2
   451→      const docX = (cx - v.panX) / v.zoom
   452→      const docY = (cy - v.panY) / v.zoom
   453→      setViewState((prev) => ({ ...prev, zoom: target, panX: cx - docX * target, panY: cy - docY * target }))
   454→    },
   455→    [viewport],
   456→  )
   457→
   458→  const zoomBy = useCallback(
   459→    (factor: number) => {
   460→      setZoom(viewRef.current.zoom * factor)
   461→    },
   462→    [setZoom],
   463→  )
   464→
   465→  const renderThumb = useCallback((p: PageModel, width: number) => renderPageThumbnail(docRef.current, p, width), [])
   466→
   467→  /* ------------------------- undo / redo ------------------------- */
   468→
   469→  const undo = useCallback(() => {
   470→    const restored = history.undo() as DesignDoc | null
   471→    if (!restored) return
   472→    docRef.current = restored
   473→    dirtyRef.current = true
   474→    setDoc(restored)
   475→    onDocChange(restored)
   476→    syncHistoryState()
   477→    setSelectedIds((prev) => prev.filter((id) => restored.pages.some((p) => p.elements.some((e) => e.id === id))))
   478→  }, [history, onDocChange, syncHistoryState])
   479→
   480→  const redo = useCallback(() => {
   481→    const restored = history.redo() as DesignDoc | null
   482→    if (!restored) return
   483→    docRef.current = restored
   484→    dirtyRef.current = true
   485→    setDoc(restored)
   486→    onDocChange(restored)
   487→    syncHistoryState()
   488→    setSelectedIds((prev) => prev.filter((id) => restored.pages.some((p) => p.elements.some((e) => e.id === id))))
   489→  }, [history, onDocChange, syncHistoryState])
   490→
   491→  /* ------------------------- api object ------------------------- */
   492→
   493→  const api = useMemo<CanvasApi>(
   494→    () => ({
   495→      doc,
   496→      page,
   497→      pageIndex,
   498→      canEdit,
   499→      selectedIds,
   500→      selection,
   501→      addElements,
   502→      updateElements,
   503→      deleteElements,
   504→      duplicateElements,
   505→      selectIds,
   506→      reorder,
   507→      groupSelected,
   508→      ungroupSelected,
   509→      renameElement,
   510→      copySelection,
   511→      pasteClipboard,
   512→      setPageIndex,
   513→      addPage,
   514→      deletePage,
   515→      movePage,
   516→      renamePage,
   517→      setPageBackground,
   518→      view,
   519→      setView,
   520→      fitToScreen,
   521→      setZoom,
   522→      undo,
   523→      redo,
   524→      canUndo: historyState.canUndo,
   525→      canRedo: historyState.canRedo,
   526→      startInlineEdit: (id: string) => setEditingId(id),
   527→      renderThumb,
   528→      dropPos,
   529→    }),
   530→    [
   531→      doc, page, pageIndex, canEdit, selectedIds, selection, addElements, updateElements, deleteElements,
   532→      duplicateElements, selectIds, reorder, groupSelected, ungroupSelected, renameElement, copySelection,
   533→      pasteClipboard, setPageIndex, addPage, deletePage, movePage, renamePage, setPageBackground, view,
   534→      setView, fitToScreen, setZoom, undo, redo, historyState, renderThumb, dropPos,
   535→    ],
   536→  )
   537→
   538→  /* ------------------------- keyboard ------------------------- */
   539→
   540→  useEffect(() => {
   541→    function isTextEntry(t: EventTarget | null): boolean {
   542→      const el = t as HTMLElement | null
   543→      return !!el && (el.tagName === "INPUT" || el.tagName === "TEXTAREA" || el.isContentEditable)
   544→    }
   545→    function onKey(e: KeyboardEvent) {
   546→      if (isTextEntry(e.target)) return
   547→      const mod = e.ctrlKey || e.metaKey
   548→      const key = e.key
   549→      if (key === "Escape") {
   550→        if (contextMenu) setContextMenu(null)
   551→        else if (activeTab) setActiveTab(null)
   552→        else setSelectedIds([])
   553→        return
   554→      }
   555→      if (!canEdit) {
   556→        if (mod && key.toLowerCase() === "a") {
   557→          e.preventDefault()
   558→          setSelectedIds(page.elements.filter((el) => !el.locked).map((el) => el.id))
   559→        }
   560→        return
   561→      }
   562→      // clipboard & history & tools
   563→      if (mod && key.toLowerCase() === "z" && !e.shiftKey) { e.preventDefault(); undo(); return }
   564→      if (mod && (key.toLowerCase() === "y" || (key.toLowerCase() === "z" && e.shiftKey))) { e.preventDefault(); redo(); return }
   565→      if (mod && key.toLowerCase() === "c") { e.preventDefault(); copySelection(false); return }
   566→      if (mod && key.toLowerCase() === "x") { e.preventDefault(); copySelection(true); return }
   567→      if (mod && key.toLowerCase() === "v") { e.preventDefault(); pasteClipboard(); return }
   568→      if (mod && key.toLowerCase() === "d") { e.preventDefault(); duplicateElements(selectedIds); return }
   569→      if (mod && key.toLowerCase() === "a") { e.preventDefault(); selectIds(page.elements.map((el) => el.id)); return }
   570→      if (mod && key.toLowerCase() === "g" && !e.shiftKey) { e.preventDefault(); groupSelected(); return }
   571→      if (mod && key.toLowerCase() === "g" && e.shiftKey) { e.preventDefault(); ungroupSelected(); return }
   572→      if (mod && key === "]") { e.preventDefault(); reorder(selectedIds, "front"); return }
   573→      if (mod && key === "[") { e.preventDefault(); reorder(selectedIds, "back"); return }
   574→      if (key === "]") { reorder(selectedIds, "forward"); return }
   575→      if (key === "[") { reorder(selectedIds, "backward"); return }
   576→      if (key === "Delete" || key === "Backspace") { if (selectedIds.length > 0) { e.preventDefault(); deleteElements(selectedIds) } return }
   577→      // nudging
   578→      if (key.startsWith("Arrow") && selectedIds.length > 0) {
   579→        e.preventDefault()
   580→        const step = e.shiftKey ? 10 : 1
   581→        const dx = key === "ArrowLeft" ? -step : key === "ArrowRight" ? step : 0
   582→        const dy = key === "ArrowUp" ? -step : key === "ArrowDown" ? step : 0
   583→        updateElements(
   584→          selection.map((el) => ({ id: el.id, patch: { x: el.x + dx, y: el.y + dy } })),
   585→          { coalesceKey: "nudge" },
   586→        )
   587→        return
   588→      }
   589→      // zoom
   590→      if (key === "+" || key === "=") { zoomBy(1.15); return }
   591→      if (key === "-" || key === "_") { zoomBy(1 / 1.15); return }
   592→      if (key === "0") { fitToScreen(); return }
   593→      if (key === "1") { setZoom(1); return }
   594→    }
   595→    window.addEventListener("keydown", onKey)
   596→    return () => window.removeEventListener("keydown", onKey)
   597→  }, [
   598→    activeTab, canEdit, contextMenu, deleteElements, duplicateElements, fitToScreen, groupSelected,
   599→    page.elements, pasteClipboard, redo, reorder, copySelection, selection, selectIds, selectedIds,
   600→    setZoom, undo, ungroupSelected, updateElements, zoomBy,
   601→  ])
   602→
   603→  /* ------------------------- editor handle ------------------------- */
   604→
   605→  useEffect(() => {
   606→    const handle: EditorHandle = {
   607→      export: async (req): Promise<ExportResult[]> => exportDoc(docRef.current, req),
   608→      getThumbnail: async () => {
   609→        const current = docRef.current
   610→        const p = current.pages[Math.min(pageIndexRef.current, current.pages.length - 1)]
   611→        return renderThumbnail(current, p)
   612→      },
   613→      isDirty: () => dirtyRef.current,
   614→    }
   615→    registerHandle(handle)
   616→    return () => registerHandle(null)
   617→  }, [registerHandle])
   618→
   619→  /* keep refs in sync */
   620→  useEffect(() => {
   621→    pageIndexRef.current = pageIndex
   622→  }, [pageIndex])
   623→
   624→  const showPagesStrip = view.pagesOpen || doc.pages.length > 1
   625→
   626→  return (
   627→    <div className="flex h-full w-full overflow-hidden bg-background">
   628→      {/* ---------------- left rail ---------------- */}
   629→      <nav
   630→        className="z-40 flex shrink-0 flex-row items-center gap-0.5 overflow-x-auto border-b bg-card px-1.5 py-1 pb-[max(0.25rem,env(safe-area-inset-bottom))] md:flex-col md:overflow-visible md:border-b-0 md:border-r md:px-1.5 md:py-2"
   631→        aria-label="Editor tools"
   632→      >
   633→        {TABS.map((tab) => (
   634→          <IconBtn
   635→            key={tab.id}
   636→            label={tab.label}
   637→            active={activeTab === tab.id}
   638→            disabled={!canEdit && (tab.id === "elements" || tab.id === "text" || tab.id === "photos")}
   639→            className="h-11 w-11"
   640→            onClick={() => setActiveTab((prev) => (prev === tab.id ? null : tab.id))}
   641→          >
   642→            {tab.icon}
   643→          </IconBtn>
   644→        ))}
   645→        <div className="mx-1 h-6 w-px bg-border md:my-1 md:h-px md:w-6" />
   646→        <IconBtn label="Undo" disabled={!historyState.canUndo} onClick={undo} className="h-11 w-11">
   647→          <Undo2 className="h-5 w-5" />
   648→        </IconBtn>
   649→        <IconBtn label="Redo" disabled={!historyState.canRedo} onClick={redo} className="h-11 w-11">
   650→          <Redo2 className="h-5 w-5" />
   651→        </IconBtn>
   652→        <div className="mx-1 h-6 w-px bg-border md:my-1 md:h-px md:w-6" />
   653→        <IconBtn
   654→          label={view.pagesOpen ? "Hide pages" : "Show pages"}
   655→          active={view.pagesOpen}
   656→          className="h-11 w-11"
   657→          onClick={() => setView({ pagesOpen: !view.pagesOpen })}
   658→        >
   659→          <Files className="h-5 w-5" />
   660→        </IconBtn>
   661→        <ViewOptions api={api} />
   662→        {/* mobile: properties trigger */}
   663→        <div className="lg:hidden">
   664→          <IconBtn label="Properties" active={propsSheetOpen} className="h-11 w-11" onClick={() => setPropsSheetOpen((v) => !v)}>
   665→            <SlidersHorizontal className="h-5 w-5" />

   671→      {activeTab !== null ? (

   673→          className="absolute inset-y-0 left-[52px] z-30 flex w-[300px] max-w-[86vw] flex-col border-r bg-card shadow-2xl md:static md:left-auto md:z-auto md:shadow-none"

   836→/* ------------------------------ rulers ------------------------------ */
   837→
   838→function Ruler({ axis, zoom, pan, docSize, size }: { axis: "h" | "v"; zoom: number; pan: number; docSize: number; size: number }) {
   839→  const canvasRef = useRef<HTMLCanvasElement>(null)
   840→  useEffect(() => {
   841→    const canvas = canvasRef.current
   842→    if (!canvas) return
   843→    const thickness = 22
   844→    const dpr = Math.min(2, window.devicePixelRatio || 1)
   845→    if (axis === "h") {
   846→      canvas.width = Math.max(1, size * dpr)
   847→      canvas.height = thickness * dpr
   848→    } else {
   849→      canvas.width = thickness * dpr
   850→      canvas.height = Math.max(1, size * dpr)
   851→    }
   852→    canvas.style.width = axis === "h" ? `${size}px` : `${thickness}px`
   853→    canvas.style.height = axis === "h" ? `${thickness}px` : `${size}px`
   854→    const ctx = canvas.getContext("2d")
   855→    if (!ctx) return
   856→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   857→    ctx.clearRect(0, 0, size, thickness)
   858→    const styles = getComputedStyle(canvas)
   859→    const fg = styles.color || "#6b7280"
   860→    ctx.strokeStyle = fg
   861→    ctx.fillStyle = fg
   862→    ctx.font = "9px Inter, sans-serif"
   863→    // choose a tick step that gives ≥ 56px spacing
   864→    const steps = [1, 2, 5, 10, 25, 50, 100, 250, 500, 1000, 2500, 5000]
   865→    const step = steps.find((s) => s * zoom >= 56) ?? 10000
   866→    const startDoc = Math.max(0, Math.floor((0 - pan) / zoom / step) * step)
   867→    const endDoc = Math.min(docSize, Math.ceil((size - pan) / zoom / step) * step)
   868→    for (let d = startDoc; d <= endDoc; d += step) {
   869→      const s = d * zoom + pan
   870→      ctx.beginPath()
   871→      if (axis === "h") {
   872→        ctx.moveTo(s, thickness - 8)
   873→        ctx.lineTo(s, thickness)
   874→        ctx.fillText(String(d), s + 3, 9)
   875→      } else {
   876→        ctx.moveTo(thickness - 8, s)
   877→        ctx.lineTo(thickness, s)
   878→        ctx.save()
   879→        ctx.translate(9, s - 3)
   880→        ctx.rotate(-Math.PI / 2)
   881→        ctx.fillText(String(d), 0, 0)
   882→        ctx.restore()
   883→      }
   884→      ctx.stroke()
   885→    }
   886→  }, [axis, zoom, pan, docSize, size])
   887→  return <canvas ref={canvasRef} className="block bg-card text-muted-foreground" role="img" aria-label={`${axis === "h" ? "Horizontal" : "Vertical"} ruler in pixels`} />
   888→}
