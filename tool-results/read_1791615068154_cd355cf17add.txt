   450→   450→              start,

   452→   452→              outPoint: kind === "image" ? duration : Math.min(duration, Math.round(info.duration) || duration),
   453→   453→              name: file.name.replace(/\.[^.]+$/, "").slice(0, 40) || kind,
   454→   454→            }),
   455→   455→          )
   456→   456→        }
   457→   457→        if (created.length) {
   458→   458→          commit({ ...d, config: { ...cfg, clips: [...cfg.clips, ...created] } })
   459→   459→          toast({ title: `Imported ${created.length} file${created.length > 1 ? "s" : ""}`, description: "Media lives in this browser session." })
   460→   460→        }
   461→   461→      } finally {
   462→   462→        setImporting(false)
   463→   463→      }
   464→   464→    },
   465→   465→    [canEdit, commit, toast],
   466→   466→  )
   467→   467→
   468→   468→  const openPicker = useCallback((relinkClipId: string | null) => {
   469→   469→    relinkTargetRef.current = relinkClipId
   470→   470→    fileInputRef.current?.click()
   471→   471→  }, [])
   472→   472→
   473→   473→  /* ---------------- transport ---------------- */
   474→   474→
   475→   475→  const seek = useCallback((ms: number) => {
   476→   476→    playheadRef.current = Math.max(0, Math.round(ms))
   477→   477→    setPlayheadMs(playheadRef.current)
   478→   478→  }, [])
   479→   479→
   480→   480→  const onTimeUpdate = useCallback((ms: number) => setPlayheadMs(ms), [])
   481→   481→
   482→   482→  const handleEnded = useCallback(() => {
   483→   483→    setPlaying(false)
   484→   484→    setPlayheadMs(playheadRef.current)
   485→   485→  }, [])
   486→   486→
   487→   487→  const togglePlay = useCallback(() => {
   488→   488→    if (playing) {
   489→   489→      setPlaying(false)
   490→   490→      setPlayheadMs(playheadRef.current)
   491→   491→      return
   492→   492→    }
   493→   493→    const cfgTotal = totalDurationMs(ensureVideoConfig(docRef.current))
   494→   494→    if (cfgTotal <= 0) return
   495→   495→    if (playheadRef.current >= cfgTotal) playheadRef.current = 0
   496→   496→    const actx = pool.ensureAudioContext()
   497→   497→    if (actx && actx.state === "suspended") void actx.resume().catch(() => {})
   498→   498→    setPlayheadMs(playheadRef.current)
   499→   499→    setPlaying(true)
   500→   500→  }, [playing, pool])
   501→   501→
   502→   502→  const stopPlayback = useCallback(() => {
   503→   503→    setPlaying(false)
   504→   504→    playheadRef.current = 0
   505→   505→    setPlayheadMs(0)
   506→   506→  }, [])
   507→   507→
   508→   508→  /* ---------------- keyboard ---------------- */
   509→   509→
   510→   510→  useEffect(() => {
   511→   511→    function isTextEntry(el: EventTarget | null): boolean {
   512→   512→      const t = el as HTMLElement | null
   513→   513→      return !!t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   514→   514→    }
   515→   515→    const onKey = (e: KeyboardEvent) => {
   516→   516→      if (e.code === "Space" && !isTextEntry(e.target)) {
   517→   517→        e.preventDefault()
   518→   518→        togglePlay()
   519→   519→        return
   520→   520→      }
   521→   521→      if (isTextEntry(e.target)) return
   522→   522→      const meta = e.ctrlKey || e.metaKey
   523→   523→      if (meta && e.key.toLowerCase() === "z") {
   524→   524→        e.preventDefault()
   525→   525→        if (e.shiftKey) redo()
   526→   526→        else undo()
   527→   527→      } else if (meta && e.key.toLowerCase() === "y") {
   528→   528→        e.preventDefault()
   529→   529→        redo()
   530→   530→      } else if ((e.key === "Delete" || e.key === "Backspace") && selectedId && canEdit) {
   531→   531→        e.preventDefault()
   532→   532→        deleteSelected()
   533→   533→      }
   534→   534→    }
   535→   535→    window.addEventListener("keydown", onKey)
   536→   536→    return () => window.removeEventListener("keydown", onKey)
   537→   537→  }, [canEdit, deleteSelected, redo, selectedId, togglePlay, undo])
   538→   538→
   539→   539→  /* ---------------- editor handle (export / thumbnail) ---------------- */
   540→   540→
   541→   541→  useEffect(() => {
   542→   542→    const handle: EditorHandle = {
   543→   543→      export: async (req): Promise<ExportResult[]> => {
   544→   544→        const base = req.filenameBase || "video"
   545→   545→        const d = docRef.current
   546→   546→        if (req.format === "json") {
   547→   547→          return [{ filename: `${base}.studio.json`, blob: new Blob([JSON.stringify(d, null, 2)], { type: "application/json" }) }]
   548→   548→        }
   549→   549→        if (req.format === "png") {
   550→   550→          const cfg = ensureVideoConfig(d)
   551→   551→          const blob = await exportCurrentFramePng(cfg, pool, playheadRef.current)
   552→   552→          return [{ filename: `${base}.png`, blob, note: "Current frame at full canvas size." }]
   553→   553→        }
   554→   554→        if (req.format === "webm" || req.format === "mp4") {
   555→   555→          if (req.format === "mp4" && !(typeof MediaRecorder !== "undefined" && MediaRecorder.isTypeSupported("video/mp4"))) {
   556→   556→            throw new Error("MP4 recording isn't supported by this browser. WebM export is available in the same menu and plays in Chrome, Edge and Firefox.")
   557→   557→          }
   558→   558→          const mime = pickVideoMime(req.format)
   559→   559→          if (!mime) throw new Error("This browser can't record video (MediaRecorder unavailable). Try Chrome, Edge or Firefox.")
   560→   560→          exportingRef.current = true
   561→   561→          setPlaying(false)
   562→   562→          seek(0)
   563→   563→          pool.pauseAll()
   564→   564→          cancelExportRef.current = false
   565→   565→          setExporting({ active: true, progress: 0 })
   566→   566→          try {
   567→   567→            const cfg = ensureVideoConfig(d)
   568→   568→            const { blob, note } = await recordTimeline({
   569→   569→              config: cfg,
   570→   570→              pool,
   571→   571→              mimeType: mime,
   572→   572→              onProgress: (p) => setExporting({ active: true, progress: p }),
   573→   573→              isCancelled: () => cancelExportRef.current,
   574→   574→            })
   575→   575→            if (!blob) return []
   576→   576→            return [
   577→   577→              {
   578→   578→                filename: `${base}.${req.format}`,
   579→   579→                blob,
   580→   580→                note: note ?? `Realtime render — export took about ${fmtClock(totalDurationMs(cfg))} (one pass over the timeline).`,
   581→   581→              },
   582→   582→            ]
   583→   583→          } finally {
   584→   584→            exportingRef.current = false
   585→   585→            setExporting({ active: false, progress: 0 })
   586→   586→          }
   587→   587→        }
   588→   588→        throw new Error(`"${req.format}" isn't available for video projects — use WebM, MP4, PNG or JSON.`)
   589→   589→      },
   590→   590→      getThumbnail: async () => {
   591→   591→        try {
   592→   592→          return await renderThumbnail(ensureVideoConfig(docRef.current), pool, playheadRef.current)
   593→   593→        } catch {
   594→   594→          return null
   595→   595→        }
   596→   596→      },
   597→   597→      isDirty: () => dirtyRef.current,
   598→   598→    }
   599→   599→    registerHandle(handle)
   600→   600→    return () => registerHandle(null)
   601→   601→  }, [registerHandle, pool, seek])
   602→   602→
   603→   603→  /* ---------------- render ---------------- */
   604→   604→
   605→   605→  const canvasPresetValue = useMemo(() => {
   606→   606→    const match = ASPECT_PRESETS.find((p) => p.width === config.canvas.width && p.height === config.canvas.height)
   607→   607→    return match ? match.id : "custom"
   608→   608→  }, [config.canvas.width, config.canvas.height])
   609→   609→
   610→   610→  return (
   611→   611→    <div className="flex h-full min-h-0 flex-col bg-muted/30">
   612→   612→      {/* top toolbar */}
   613→   613→      <div className="flex flex-wrap items-center gap-1.5 border-b bg-card px-2 py-1.5">
   614→   614→        <Button size="sm" variant="outline" className="h-8 gap-1.5" disabled={!canEdit || importing} onClick={() => openPicker(null)}>
   615→   615→          {importing ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Upload className="h-3.5 w-3.5" />}
   616→   616→          Import media
   617→   617→        </Button>
   618→   618→        <Button size="sm" variant="outline" className="h-8 gap-1.5" disabled={!canEdit} onClick={() => addTextClip("text")}>
   619→   619→          <TypeIcon className="h-3.5 w-3.5" /> Text
   620→   620→        </Button>
   621→   621→        <div className="flex items-center gap-1.5">
   622→   622→          <MonitorPlay className="h-3.5 w-3.5 text-muted-foreground" />
   623→   623→          <Select value={canvasPresetValue} disabled={!canEdit} onValueChange={applyAspect}>
   624→   624→            <SelectTrigger className="h-8 w-[130px] text-xs" aria-label="Canvas aspect ratio">
   625→   625→              <SelectValue placeholder="Aspect" />
   626→   626→            </SelectTrigger>
   627→   627→            <SelectContent>
   628→   628→              {ASPECT_PRESETS.map((p) => (
   629→   629→                <SelectItem key={p.id} value={p.id}>
   630→   630→                  {p.label} · {p.width}×{p.height}
   631→   631→                </SelectItem>
   632→   632→              ))}
   633→   633→              {canvasPresetValue === "custom" && (
   634→   634→                <SelectItem value="custom">
   635→   635→                  {config.canvas.width}×{config.canvas.height}
   636→   636→                </SelectItem>
   637→   637→              )}
   638→   638→            </SelectContent>
   639→   639→          </Select>
   640→   640→        </div>
   641→   641→        <div className="ml-auto flex items-center gap-1">
   642→   642→          <span className="mr-1 hidden items-center gap-1 text-[11px] text-muted-foreground sm:flex">
   643→   643→            <Clock className="h-3 w-3" /> {fmtClock(total)}
   644→   644→          </span>
   645→   645→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={undo} disabled={!historyState.canUndo} aria-label="Undo" title="Undo (Ctrl+Z)">
   646→   646→            <Undo2 className="h-4 w-4" />
   647→   647→          </Button>
   648→   648→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={redo} disabled={!historyState.canRedo} aria-label="Redo" title="Redo (Ctrl+Y)">
   649→   649→            <Redo2 className="h-4 w-4" />
   650→   650→          </Button>
   651→   651→        </div>
   652→   652→      </div>
   653→   653→
   654→   654→      {/* session media notice */}
   655→   655→      {hasMedia && (
   656→   656→        <div className="flex items-center gap-1.5 border-b bg-amber-500/10 px-3 py-1 text-[11px] text-amber-900 dark:text-amber-200">
   657→   657→          <Info className="h-3 w-3 shrink-0" />
   658→   658→          <span>
   659→   659→            Media files live in this browser session; re-link after reload.
   660→   660→            {deadCount > 0 && ` ${deadCount} clip${deadCount > 1 ? "s" : ""} need${deadCount === 1 ? "s" : ""} relinking.`}
   661→   661→          </span>
   662→   662→        </div>
   663→   663→      )}
   664→   664→
   665→   665→      <input
   666→   666→        ref={fileInputRef}
   667→   667→        type="file"
   668→   668→        multiple
   669→   669→        accept="video/*,audio/*,image/*"
   670→   670→        className="hidden"
   671→   671→        aria-hidden
   672→   672→        onChange={(e) => {
   673→   673→          void handleFiles(e.target.files)
   674→   674→          e.target.value = ""
   675→   675→        }}
   676→   676→      />
   677→   677→
   678→   678→      <div className="flex min-h-0 flex-1 flex-col lg:flex-row">
   679→   679→        {/* preview */}
   680→   680→        <div className="relative flex min-h-[260px] min-w-0 flex-1 flex-col lg:flex-[1.5]">
   681→   681→          <PreviewPlayer
   682→   682→            config={config}
   683→   683→            playing={playing}
   684→   684→            suspended={exporting.active}
   685→   685→            playheadRef={playheadRef}
   686→   686→            playheadMs={playheadMs}
   687→   687→            pool={pool}
   688→   688→            deadCount={deadCount}
   689→   689→            onTogglePlay={togglePlay}
   690→   690→            onStop={stopPlayback}
   691→   691→            onEnded={handleEnded}
   692→   692→            onTimeUpdate={onTimeUpdate}
   693→   693→          />
   694→   694→          {exporting.active && (
   695→   695→            <div className="absolute inset-0 z-30 flex flex-col items-center justify-center gap-3 bg-background/85 p-6 backdrop-blur-sm">
   696→   696→              <Film className="h-5 w-5 animate-pulse text-primary" />
   697→   697→              <p className="text-sm font-semibold">Rendering video in real time…</p>
   698→   698→              <Progress value={Math.round(exporting.progress * 100)} className="w-64" />
   699→   699→              <p className="text-center text-xs text-muted-foreground">
   700→   700→                {Math.round(exporting.progress * 100)}% · keep this tab visible until it finishes
   701→   701→              </p>
   702→   702→              <Button
   703→   703→                variant="outline"
   704→   704→                size="sm"
   705→   705→                onClick={() => {
   706→   706→                  cancelExportRef.current = true
   707→   707→                }}
   708→   708→              >
   709→   709→                Cancel export
   710→   710→              </Button>
   711→   711→            </div>
   712→   712→          )}
   713→   713→        </div>
   714→   714→
   715→   715→        {/* properties */}
   716→   716→        <div className={cn("flex min-h-0 w-full shrink-0 flex-col border-t bg-card lg:w-80 lg:border-l lg:border-t-0")}>
   717→   717→          <PropertiesPanel
   718→   718→            config={config}
   719→   719→            selected={selected}
   720→   720→            canEdit={canEdit}
   721→   721→            playheadMs={playheadMs}
   722→   722→            onPatchClip={patchClip}
   723→   723→            onAddText={() => addTextClip("text")}
   724→   724→            onAddSticker={(char) => addTextClip("sticker", char)}
   725→   725→            onAddSubtitle={addSubtitle}
   726→   726→            onPatchSubtitle={patchSubtitle}
   727→   727→            onDeleteSubtitle={deleteSubtitle}
   728→   728→            onImportSrt={importSrt}
   729→   729→            onDownloadSrt={downloadSrt}
   730→   730→            isDead={(clipId) => deadSet.has(clipId)}
   731→   731→            onRelink={(clipId) => openPicker(clipId)}
   732→   732→            onDeleteClip={deleteClipById}
   733→   733→          />
   734→   734→        </div>
   735→   735→      </div>
   736→   736→
   737→   737→      {/* timeline */}
   738→   738→      <Timeline
   739→   739→        config={config}
   740→   740→        total={total}
   741→   741→        zoom={zoom}
   742→   742→        onZoomChange={setZoom}
   743→   743→        playheadMs={playheadMs}
   744→   744→        onScrub={seek}
   745→   745→        selectedId={selectedId}
   746→   746→        onSelect={setSelectedId}
   747→   747→        canEdit={canEdit}
   748→   748→        onPatchClip={patchClip}
   749→   749→        onPatchTrack={patchTrack}
   750→   750→        onSplit={splitAtPlayhead}
   751→   751→        onDuplicate={duplicateSelected}
   752→   752→        onDelete={deleteSelected}
   753→   753→        deadIds={deadSet}
   754→   754→        onRelink={(clipId) => openPicker(clipId)}
   755→   755→        getSourceDuration={(clipId) => pool.sourceDurations.get(clipId)}
   756→   756→      />
   757→   757→    </div>
   758→   758→  )
   759→   759→})
