    92→    92→    92→  const [viewport, setViewport] = useState({ w: 0, h: 0 })

   108→   108→   108→

   112→   112→   112→
   113→   113→   113→  /* ------------------------- core mutations ------------------------- */
   114→   114→   114→
   115→   115→   115→  const commit = useCallback(
   116→   116→   116→    (next: DesignDoc, coalesceKey?: string) => {

   118→   118→   118→      dirtyRef.current = true
   119→   119→   119→      setDoc(next)
   120→   120→   120→      history.push(next, coalesceKey)
   121→   121→   121→      syncHistoryState()
   122→   122→   122→      onDocChange(next)
   123→   123→   123→    },
   124→   124→   124→    [history, onDocChange, syncHistoryState],
   125→   125→   125→  )
   126→   126→   126→
   127→   127→   127→  const mutatePage = useCallback(
   128→   128→   128→    (fn: (page: PageModel) => PageModel, coalesceKey?: string) => {
   129→   129→   129→      const prev = docRef.current
   130→   130→   130→      const next: DesignDoc = {
   131→   131→   131→        ...prev,
   132→   132→   132→        pages: prev.pages.map((p, i) => (i === pageIndexRef.current ? fn(p) : p)),
   133→   133→   133→      }
   134→   134→   134→      commit(next, coalesceKey)
   135→   135→   135→    },
   136→   136→   136→    [commit],
   137→   137→   137→  )
   138→   138→   138→
   139→   139→   139→  const ensureTextHeight = (el: DesignElement): DesignElement => {
   140→   140→   140→    if (el.type !== "text") return el
   141→   141→   141→    const t = el as TextElement
   142→   142→   142→    const h = measureTextBlockHeight(
   143→   143→   143→      {
   144→   144→   144→        text: t.text,
   145→   145→   145→        fontFamily: t.fontFamily,
   146→   146→   146→        fontSize: t.fontSize,
   147→   147→   147→        fontWeight: t.fontWeight,
   148→   148→   148→        italic: t.italic,
   149→   149→   149→        uppercase: t.uppercase,
   150→   150→   150→        lineHeight: t.lineHeight,
   151→   151→   151→        letterSpacing: t.letterSpacing,
   152→   152→   152→        listStyle: t.listStyle,
   153→   153→   153→      },
   154→   154→   154→      t.width,
   155→   155→   155→    )
   156→   156→   156→    return { ...t, height: Math.max(t.height, Math.round(h)) }
   157→   157→   157→  }
   158→   158→   158→
   159→   159→   159→  const updateElements = useCallback(
   160→   160→   160→    (patches: ElementPatch[], opts?: { coalesceKey?: string }) => {
   161→   161→   161→      if (!canEdit || patches.length === 0) return
   162→   162→   162→      const map = new Map(patches.map((p) => [p.id, p.patch]))
   163→   163→   163→      mutatePage(
   164→   164→   164→        (p) => ({
   165→   165→   165→          ...p,
   166→   166→   166→          elements: p.elements.map((el) => {
   167→   167→   167→            const patch = map.get(el.id)
   168→   168→   168→            if (!patch) return el
   169→   169→   169→            const merged = { ...el, ...patch } as DesignElement
   170→   170→   170→            const textKeys = ["text", "fontSize", "fontFamily", "fontWeight", "italic", "uppercase", "lineHeight", "letterSpacing", "listStyle", "width"]
   171→   171→   171→            if (merged.type === "text" && textKeys.some((k) => k in patch)) return ensureTextHeight(merged)
   172→   172→   172→            return merged
   173→   173→   173→          }),
   174→   174→   174→        }),
   175→   175→   175→        opts?.coalesceKey,
   176→   176→   176→      )
   177→   177→   177→    },
   178→   178→   178→    [canEdit, mutatePage],
   179→   179→   179→  )
   180→   180→   180→
   181→   181→   181→  const dropPos = useCallback(
   182→   182→   182→    (w: number, h: number) => {
   183→   183→   183→      const v = viewRef.current
   184→   184→   184→      const vp = viewport.w > 0 ? viewport : { w: 900, h: 600 }
   185→   185→   185→      const cx = (vp.w / 2 - v.panX) / v.zoom
   186→   186→   186→      const cy = (vp.h / 2 - v.panY) / v.zoom
   187→   187→   187→      return {
   188→   188→   188→        x: Math.max(12, Math.round(cx - w / 2)),
   189→   189→   189→        y: Math.max(12, Math.round(cy - h / 2)),
   190→   190→   190→      }
   191→   191→   191→    },
   192→   192→   192→    [viewport],
   193→   193→   193→  )
   194→   194→   194→
   195→   195→   195→  const addElements = useCallback(
   196→   196→   196→    (els: DesignElement[], select = true) => {
   197→   197→   197→      if (!canEdit || els.length === 0) return
   198→   198→   198→      const prepared = els.map(ensureTextHeight)
   199→   199→   199→      mutatePage((p) => ({ ...p, elements: [...p.elements, ...prepared] }))
   200→   200→   200→      if (select) setSelectedIds(prepared.map((el) => el.id))
   201→   201→   201→    },
   202→   202→   202→    [canEdit, mutatePage],
   203→   203→   203→  )
   204→   204→   204→
   205→   205→   205→  const deleteElements = useCallback(
   206→   206→   206→    (ids: string[]) => {
   207→   207→   207→      if (!canEdit || ids.length === 0) return
   208→   208→   208→      const set = new Set(ids)
   209→   209→   209→      mutatePage((p) => ({ ...p, elements: p.elements.filter((e) => !set.has(e.id)) }))
   210→   210→   210→      setSelectedIds((prev) => prev.filter((id) => !set.has(id)))
   211→   211→   211→      setEditingId(null)
   212→   212→   212→    },
   213→   213→   213→    [canEdit, mutatePage],
   214→   214→   214→  )
   215→   215→   215→
   216→   216→   216→  const duplicateElements = useCallback(
   217→   217→   217→    (ids: string[]) => {
   218→   218→   218→      if (!canEdit || ids.length === 0) return
   219→   219→   219→      const set = new Set(ids)
   220→   220→   220→      const source = page.elements.filter((e) => set.has(e.id))
   221→   221→   221→      if (source.length === 0) return
   222→   222→   222→      const gidMap = new Map<string, string>()
   223→   223→   223→      const clones = source.map((el) => {
   224→   224→   224→        let groupId = el.groupId
   225→   225→   225→        if (groupId) {
   226→   226→   226→          if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   227→   227→   227→          groupId = gidMap.get(groupId)
   228→   228→   228→        }
   229→   229→   229→        return { ...el, id: uid("dup"), x: el.x + 24, y: el.y + 24, groupId } as DesignElement
   230→   230→   230→      })
   231→   231→   231→      mutatePage((p) => ({ ...p, elements: [...p.elements, ...clones] }))
   232→   232→   232→      setSelectedIds(clones.map((c) => c.id))
   233→   233→   233→    },
   234→   234→   234→    [canEdit, mutatePage, page.elements],
   235→   235→   235→  )
   236→   236→   236→
   237→   237→   237→  const selectIds = useCallback((ids: string[], additive = false) => {
   238→   238→   238→    setEditingId(null)
   239→   239→   239→    setSelectedIds((prev) => {
   240→   240→   240→      if (!additive) return ids
   241→   241→   241→      const set = new Set(prev)
   242→   242→   242→      for (const id of ids) {
   243→   243→   243→        if (set.has(id)) set.delete(id)
   244→   244→   244→        else set.add(id)
   245→   245→   245→      }
   246→   246→   246→      return Array.from(set)
   247→   247→   247→    })
   248→   248→   248→  }, [])
   249→   249→   249→
   250→   250→   250→  const reorder = useCallback(
   251→   251→   251→    (ids: string[], dir: "front" | "forward" | "backward" | "back") => {
   252→   252→   252→      if (!canEdit || ids.length === 0) return
   253→   253→   253→      const set = new Set(ids)
   254→   254→   254→      mutatePage((p) => {
   255→   255→   255→        let elements = [...p.elements]
   256→   256→   256→        if (dir === "front") {
   257→   257→   257→          const picked = elements.filter((e) => set.has(e.id))
   258→   258→   258→          elements = [...elements.filter((e) => !set.has(e.id)), ...picked]
   259→   259→   259→        } else if (dir === "back") {
   260→   260→   260→          const picked = elements.filter((e) => set.has(e.id))
   261→   261→   261→          elements = [...picked, ...elements.filter((e) => !set.has(e.id))]
   262→   262→   262→        } else if (dir === "forward") {
   263→   263→   263→          for (let i = elements.length - 2; i >= 0; i -= 1) {
   264→   264→   264→            if (set.has(elements[i].id) && !set.has(elements[i + 1].id)) {
   265→   265→   265→              ;[elements[i], elements[i + 1]] = [elements[i + 1], elements[i]]
   266→   266→   266→            }
   267→   267→   267→          }
   268→   268→   268→        } else {
   269→   269→   269→          for (let i = 1; i < elements.length; i += 1) {
   270→   270→   270→            if (set.has(elements[i].id) && !set.has(elements[i - 1].id)) {
   271→   271→   271→              ;[elements[i], elements[i - 1]] = [elements[i - 1], elements[i]]
   272→   272→   272→            }
   273→   273→   273→          }
   274→   274→   274→        }
   275→   275→   275→        return { ...p, elements }
   276→   276→   276→      })
   277→   277→   277→    },
   278→   278→   278→    [canEdit, mutatePage],
   279→   279→   279→  )
   280→   280→   280→
   281→   281→   281→  const groupSelected = useCallback(() => {
   282→   282→   282→    if (!canEdit || selection.length < 2) return
   283→   283→   283→    const gid = uid("grp")
   284→   284→   284→    updateElements(selection.map((e) => ({ id: e.id, patch: { groupId: gid } })))
   285→   285→   285→  }, [canEdit, selection, updateElements])
   286→   286→   286→
   287→   287→   287→  const ungroupSelected = useCallback(() => {
   288→   288→   288→    if (!canEdit || selection.length === 0) return

   290→   290→   290→  }, [canEdit, selection, updateElements])
   291→   291→   291→
   292→   292→   292→  const renameElement = useCallback(

   295→   295→   295→    },

   297→   297→   297→  )

   300→   300→   300→
   301→   301→   301→  const copySelection = useCallback(

   303→   303→   303→      if (selection.length === 0) return
   304→   304→   304→      clipboard.current = { elements: JSON.parse(JSON.stringify(selection)) as DesignElement[] }
   305→   305→   305→      if (cut) deleteElements(selectedIds)
   306→   306→   306→    },
   307→   307→   307→    [deleteElements, selection, selectedIds],
   308→   308→   308→  )
   309→   309→   309→
   310→   310→   310→  const pasteClipboard = useCallback(() => {
   311→   311→   311→    const clip = clipboard.current

   313→   313→   313→    const gidMap = new Map<string, string>()
   314→   314→   314→    const clones = clip.elements.map((el) => {
   315→   315→   315→      let groupId = el.groupId
   316→   316→   316→      if (groupId) {
   317→   317→   317→        if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   318→   318→   318→        groupId = gidMap.get(groupId)
   319→   319→   319→      }
   320→   320→   320→      return { ...el, id: uid("paste"), x: el.x + 24, y: el.y + 24, groupId } as DesignElement
   321→   321→   321→    })
   322→   322→   322→    addElements(clones)
   323→   323→   323→  }, [addElements, canEdit])
   324→   324→   324→

   327→   327→   327→  const setPageIndex = useCallback((i: number) => {
   328→   328→   328→    const docCurrent = docRef.current
   329→   329→   329→    const clamped = Math.max(0, Math.min(i, docCurrent.pages.length - 1))
   330→   330→   330→    pageIndexRef.current = clamped
   331→   331→   331→    setPageIndexState(clamped)
   332→   332→   332→    setSelectedIds([])
   333→   333→   333→    setEditingId(null)
   334→   334→   334→  }, [])
   335→   335→   335→
   336→   336→   336→  const addPage = useCallback(
   337→   337→   337→    (duplicate: boolean) => {
   338→   338→   338→      if (!canEdit) return
   339→   339→   339→      const prev = docRef.current
   340→   340→   340→      const current = prev.pages[pageIndexRef.current]
   341→   341→   341→      let newPage: PageModel
   342→   342→   342→      if (duplicate) {
   343→   343→   343→        const gidMap = new Map<string, string>()
   344→   344→   344→        newPage = {
   345→   345→   345→          ...current,
   346→   346→   346→          id: uid("page"),
   347→   347→   347→          name: `${current.name} copy`,
   348→   348→   348→          elements: current.elements.map((el) => {
   349→   349→   349→            let groupId = el.groupId
   350→   350→   350→            if (groupId) {
   351→   351→   351→              if (!gidMap.has(groupId)) gidMap.set(groupId, uid("grp"))
   352→   352→   352→              groupId = gidMap.get(groupId)
   353→   353→   353→            }
   354→   354→   354→            return { ...el, id: uid("el"), groupId } as DesignElement
   355→   355→   355→          }),
   356→   356→   356→        }
   357→   357→   357→      } else {
   358→   358→   358→        newPage = createPage({ name: `Page ${prev.pages.length + 1}`, background: current.background })
   359→   359→   359→      }
   360→   360→   360→      const pages = [...prev.pages]
   361→   361→   361→      pages.splice(pageIndexRef.current + 1, 0, newPage)
   362→   362→   362→      commit({ ...prev, pages })
   363→   363→   363→      setPageIndex(pageIndexRef.current + 1)
   364→   364→   364→    },
   365→   365→   365→    [canEdit, commit, setPageIndex],
   366→   366→   366→  )
   367→   367→   367→
   368→   368→   368→  const deletePage = useCallback(
   369→   369→   369→    (index: number) => {
   370→   370→   370→      if (!canEdit) return
   371→   371→   371→      const prev = docRef.current
   372→   372→   372→      if (prev.pages.length <= 1) return
   373→   373→   373→      const pages = prev.pages.filter((_, i) => i !== index)
   374→   374→   374→      commit({ ...prev, pages })
   375→   375→   375→      setPageIndex(Math.max(0, Math.min(pageIndexRef.current > index ? pageIndexRef.current - 1 : pageIndexRef.current, pages.length - 1)))
   376→   376→   376→    },
   377→   377→   377→    [canEdit, commit, setPageIndex],
   378→   378→   378→  )
   379→   379→   379→
   380→   380→   380→  const movePage = useCallback(

   542→   542→   542→      const el = t as HTMLElement | null

   546→   546→   546→      if (isTextEntry(e.target)) return
   547→   547→   547→      const mod = e.ctrlKey || e.metaKey
   548→   548→   548→      const key = e.key
   549→   549→   549→      if (key === "Escape") {
   550→   550→   550→        if (contextMenu) setContextMenu(null)

   560→   560→   560→        return
   561→   561→   561→      }
   562→   562→   562→      // clipboard & history & tools
   563→   563→   563→      if (mod && key.toLowerCase() === "z" && !e.shiftKey) { e.preventDefault(); undo(); return }
   564→   564→   564→      if (mod && (key.toLowerCase() === "y" || (key.toLowerCase() === "z" && e.shiftKey))) { e.preventDefault(); redo(); return }
   565→   565→   565→      if (mod && key.toLowerCase() === "c") { e.preventDefault(); copySelection(false); return }
   566→   566→   566→      if (mod && key.toLowerCase() === "x") { e.preventDefault(); copySelection(true); return }
   567→   567→   567→      if (mod && key.toLowerCase() === "v") { e.preventDefault(); pasteClipboard(); return }
   568→   568→   568→      if (mod && key.toLowerCase() === "d") { e.preventDefault(); duplicateElements(selectedIds); return }
   569→   569→   569→      if (mod && key.toLowerCase() === "a") { e.preventDefault(); selectIds(page.elements.map((el) => el.id)); return }
   570→   570→   570→      if (mod && key.toLowerCase() === "g" && !e.shiftKey) { e.preventDefault(); groupSelected(); return }
   571→   571→   571→      if (mod && key.toLowerCase() === "g" && e.shiftKey) { e.preventDefault(); ungroupSelected(); return }
   572→   572→   572→      if (mod && key === "]") { e.preventDefault(); reorder(selectedIds, "front"); return }
   573→   573→   573→      if (mod && key === "[") { e.preventDefault(); reorder(selectedIds, "back"); return }
   574→   574→   574→      if (key === "]") { reorder(selectedIds, "forward"); return }
   575→   575→   575→      if (key === "[") { reorder(selectedIds, "backward"); return }
   576→   576→   576→      if (key === "Delete" || key === "Backspace") { if (selectedIds.length > 0) { e.preventDefault(); deleteElements(selectedIds) } return }

   587→   587→   587→        return
   588→   588→   588→      }

   591→   591→   591→      if (key === "-" || key === "_") { zoomBy(1 / 1.15); return }
   592→   592→   592→      if (key === "0") { fitToScreen(); return }
   593→   593→   593→      if (key === "1") { setZoom(1); return }
   594→   594→   594→    }

   596→   596→   596→    return () => window.removeEventListener("keydown", onKey)

   838→   838→   838→function Ruler({ axis, zoom, pan, docSize, size }: { axis: "h" | "v"; zoom: number; pan: number; docSize: number; size: number }) {
   839→   839→   839→  const canvasRef = useRef<HTMLCanvasElement>(null)
   840→   840→   840→  useEffect(() => {
   841→   841→   841→    const canvas = canvasRef.current
   842→   842→   842→    if (!canvas) return
   843→   843→   843→    const thickness = 22
   844→   844→   844→    const dpr = Math.min(2, window.devicePixelRatio || 1)
   845→   845→   845→    if (axis === "h") {
   846→   846→   846→      canvas.width = Math.max(1, size * dpr)
   847→   847→   847→      canvas.height = thickness * dpr
   848→   848→   848→    } else {
   849→   849→   849→      canvas.width = thickness * dpr
   850→   850→   850→      canvas.height = Math.max(1, size * dpr)
   851→   851→   851→    }
   852→   852→   852→    canvas.style.width = axis === "h" ? `${size}px` : `${thickness}px`
   853→   853→   853→    canvas.style.height = axis === "h" ? `${thickness}px` : `${size}px`
   854→   854→   854→    const ctx = canvas.getContext("2d")
   855→   855→   855→    if (!ctx) return
   856→   856→   856→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   857→   857→   857→    ctx.clearRect(0, 0, size, thickness)
   858→   858→   858→    const styles = getComputedStyle(canvas)
   859→   859→   859→    const fg = styles.color || "#6b7280"
   860→   860→   860→    ctx.strokeStyle = fg
   861→   861→   861→    ctx.fillStyle = fg
   862→   862→   862→    ctx.font = "9px Inter, sans-serif"
   863→   863→   863→    // choose a tick step that gives ≥ 56px spacing
   864→   864→   864→    const steps = [1, 2, 5, 10, 25, 50, 100, 250, 500, 1000, 2500, 5000]
   865→   865→   865→    const step = steps.find((s) => s * zoom >= 56) ?? 10000
   866→   866→   866→    const startDoc = Math.max(0, Math.floor((0 - pan) / zoom / step) * step)
   867→   867→   867→    const endDoc = Math.min(docSize, Math.ceil((size - pan) / zoom / step) * step)
   868→   868→   868→    for (let d = startDoc; d <= endDoc; d += step) {
   869→   869→   869→      const s = d * zoom + pan
   870→   870→   870→      ctx.beginPath()
   871→   871→   871→      if (axis === "h") {
   872→   872→   872→        ctx.moveTo(s, thickness - 8)
   873→   873→   873→        ctx.lineTo(s, thickness)
   874→   874→   874→        ctx.fillText(String(d), s + 3, 9)
   875→   875→   875→      } else {
   876→   876→   876→        ctx.moveTo(thickness - 8, s)
   877→   877→   877→        ctx.lineTo(thickness, s)
   878→   878→   878→        ctx.save()
   879→   879→   879→        ctx.translate(9, s - 3)
   880→   880→   880→        ctx.rotate(-Math.PI / 2)
   881→   881→   881→        ctx.fillText(String(d), 0, 0)
   882→   882→   882→        ctx.restore()
   883→   883→   883→      }
   884→   884→   884→      ctx.stroke()
   885→   885→   885→    }
   886→   886→   886→  }, [axis, zoom, pan, docSize, size])
   887→   887→   887→  return <canvas ref={canvasRef} className="block bg-card text-muted-foreground" role="img" aria-label={`${axis === "h" ? "Horizontal" : "Vertical"} ruler in pixels`} />
   888→   888→   888→}
