   112→export function txt(s: string, o: TxtOpts): TextElement {
   113→  const lh = o.lh ?? 1.2
   114→  const lines = s.split("\n").length
   115→  return createText({
   116→    text: s,
   117→    x: o.x,
   118→    y: o.y,
   119→    width: o.w ?? 600,
   120→    height: o.h ?? Math.ceil(lines * o.size * lh) + 6,
   121→    fontFamily: o.font ?? F.sans,
   122→    fontSize: o.size,
   123→    fontWeight: o.weight ?? 700,
   124→    color: o.color ?? "#111827",
   125→    align: o.align ?? "left",
   126→    vAlign: o.vAlign ?? "top",
   127→    lineHeight: lh,
   128→    letterSpacing: o.ls ?? 0,
   129→    italic: o.italic ?? false,
   130→    uppercase: o.upper ?? false,
   131→    opacity: o.opacity ?? 1,
   132→    rotation: o.rotation ?? 0,
   133→    bgColor: o.bgColor,
   134→  })
   135→}

   298→/* ------------------------------ layout helpers ------------------------------ */

   300→/** x position to horizontally center a box of width w in a doc of width docW. */
   301→export function centerX(docW: number, w: number): number {
   302→  return Math.round((docW - w) / 2)
   303→}
   304→
   305→/** y position to vertically center a box of height h in a doc of height docH. */
   306→export function centerY(docH: number, h: number): number {
   307→  return Math.round((docH - h) / 2)
   308→}
   309→
   310→/** Position elements in a vertical stack starting at (x, y) with fixed gap. Mutates and returns the elements. */
   311→export function vstack(x: number, y: number, gap: number, els: DesignElement[]): DesignElement[] {
   312→  let cursor = y
   313→  for (const el of els) {
   314→    el.x = x
   315→    el.y = cursor
   316→    cursor += el.height + gap
   317→  }
   318→  return els
   319→}
   320→
   321→/** Position elements in a horizontal row starting at (x, y) with fixed gap. Mutates and returns the elements. */
   322→export function hstack(x: number, y: number, gap: number, els: DesignElement[]): DesignElement[] {
   323→  let cursor = x
   324→  for (const el of els) {
   325→    el.x = cursor
   326→    el.y = y
   327→    cursor += el.width + gap
   328→  }
   329→  return els
   330→}
   331→
   332→/** Position elements in a row-major grid (cols wide, fixed cell size + gaps). Mutates and returns the elements. */
   333→export function grid(
   334→  x: number,
   335→  y: number,
   336→  cols: number,
   337→  cellW: number,
   338→  cellH: number,
   339→  gapX: number,
   340→  gapY: number,
   341→  els: DesignElement[],
   342→): DesignElement[] {
   343→  els.forEach((el, i) => {
   344→    const col = i % cols
   345→    const row = Math.floor(i / cols)
   346→    el.x = x + col * (cellW + gapX)
   347→    el.y = y + row * (cellH + gapY)
   348→  })
   349→  return els
   350→}

   352→/** Rounded "pill" button/label — returns [background shape, centered label]. */
   353→export function pill(
   354→  label: string,
   355→  o: { x: number; y: number; w: number; h?: number; fill?: string; color?: string; size?: number; font?: string; weight?: number },
   356→): DesignElement[] {
   357→  const h = o.h ?? Math.round((o.size ?? 36) * 2)
   358→  return [
   359→    rect({ x: o.x, y: o.y, w: o.w, h, fill: o.fill ?? "#7c3aed", r: h / 2 }),
   360→    txt(label, {
   361→      x: o.x,
   362→      y: o.y,
   363→      w: o.w,
   364→      h,
   365→      size: o.size ?? 36,
   366→      font: o.font ?? F.sans,
   367→      weight: o.weight ?? 600,
   368→      color: o.color ?? "#ffffff",
   369→      align: "center",
   370→      vAlign: "middle",
   371→    }),
   372→  ]
   373→}

   375→/** Thin horizontal rule. */
   376→export function rule(x: number, y: number, w: number, color: string, thickness = 6, dash?: number[]): ShapeElement {
   377→  return rect({ x, y, w, h: thickness, fill: color, r: thickness / 2, dash })
