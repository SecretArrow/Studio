     1→"use client"
     2→
     3→/**
     4→ * StageView — the interactive Konva canvas.
     5→ *
     6→ *  • pan (space+drag, middle-drag, wheel, two-finger touch) & zoom (ctrl+wheel,
     7→ *    pinch, buttons via api.setView)
     8→ *  • click / shift-click / rubber-band multi-select, group-aware selection
     9→ *  • dragging with snapping (page center/edges, element edges+centers, grid)
    10→ *    and live guide lines (imperatively drawn — no setState in the hot path)
    11→ *  • rotate/resize via Transformer, group members move together
    12→ *  • double-click inline text editing (overlay textarea)
    13→ *  • right-click & long-press context menu trigger
    14→ */
    15→

    39→const MIN_ZOOM = 0.04
    40→const MAX_ZOOM = 8
    41→
    42→interface StageViewProps {
    43→  api: CanvasApi
    44→  editingId: string | null
    45→  onViewportResize: (w: number, h: number) => void
    46→  onContextMenu: (screenX: number, screenY: number, elementId: string | null) => void
    47→}
    48→
    49→interface DragState {
    50→  id: string
    51→  groupId?: string
    52→  orig: Map<string, { x: number; y: number }>

    55→export function StageView({ api, editingId, onViewportResize, onContextMenu }: StageViewProps) {
    56→  const wrapRef = useRef<HTMLDivElement>(null)
    57→  const stageRef = useRef<Konva.Stage>(null)
    58→  const contentLayerRef = useRef<Konva.Layer>(null)
    59→  const overlayLayerRef = useRef<Konva.Layer>(null)
    60→  const trRef = useRef<Konva.Transformer>(null)
    61→  const marqueeRef = useRef<Konva.Rect>(null)
    62→
    63→  const [size, setSize] = useState({ w: 800, h: 600 })
    64→  const [editingRect, setEditingRect] = useState<{ left: number; top: number; width: number; height: number } | null>(null)
    65→
    66→  const dragState = useRef<DragState | null>(null)
    67→  const panState = useRef<{ active: boolean; lastX: number; lastY: number } | null>(null)
    68→  const spaceDown = useRef(false)
    69→  const marquee = useRef<{ active: boolean; startX: number; startY: number; additive: boolean } | null>(null)
    70→  const longPress = useRef<{ timer: ReturnType<typeof setTimeout> | null; x: number; y: number; targetId: string | null } | null>(null)
    71→  const gestureCommit = useRef<ReturnType<typeof setTimeout> | null>(null)
    72→  const viewRef = useRef(api.view)
    73→  useEffect(() => {
    74→    viewRef.current = api.view
    75→  }, [api.view])
    76→
    77→  const { doc, page, canEdit, selectedIds } = api
    78→  const elements = page.elements
    79→  const images = useProcessedImages(elements, page.background)
    80→  const qrImages = useQrImages(elements)
    81→
    82→  /* ------------------------- container sizing ------------------------- */
    83→  useEffect(() => {
    84→    const wrap = wrapRef.current
    85→    if (!wrap) return
    86→    const ro = new ResizeObserver((entries) => {
    87→      const r = entries[0]?.contentRect
    88→      if (r) {
    89→        setSize({ w: Math.max(100, r.width), h: Math.max(100, r.height) })
    90→        onViewportResize(r.width, r.height)
    91→      }
    92→    })
    93→    ro.observe(wrap)
    94→    return () => ro.disconnect()
    95→  }, [onViewportResize])
    96→
    97→  /* ------------------------- guides (imperative) ------------------------- */
    98→  const drawGuides = useCallback((guides: GuideLine[]) => {
    99→    const layer = overlayLayerRef.current
   100→    if (!layer) return
   101→    layer.destroyChildren()
   102→    for (const g of guides) {
   103→      const line = new Konva.Line({
   104→        points: g.axis === "v" ? [g.pos, g.from, g.pos, g.to] : [g.from, g.pos, g.to, g.pos],
   105→        stroke: "#ec4899",
   106→        strokeWidth: 1 / (viewRef.current.zoom || 1),
   107→        dash: [4 / (viewRef.current.zoom || 1), 4 / (viewRef.current.zoom || 1)],
   108→        listening: false,
   109→      })
   110→      layer.add(line)
   111→    }
   112→    layer.batchDraw()
   113→  }, [])
   114→
   115→  const clearGuides = useCallback(() => {
   116→    const layer = overlayLayerRef.current
   117→    if (!layer) return
   118→    layer.destroyChildren()
   119→    layer.batchDraw()
   120→  }, [])
   121→
   122→  /* ------------------------- transformer sync ------------------------- */
   123→  useEffect(() => {
   124→    const layer = contentLayerRef.current
   125→    const tr = trRef.current
   126→    if (!layer || !tr) return
   127→    const nodes = layer.getChildren((n: Konva.Node) => n.getAttr("elementId") && selectedIds.includes(String(n.getAttr("elementId"))))
   128→    tr.nodes(nodes as Konva.Node[])
   129→    tr.getLayer()?.batchDraw()
   130→  }, [selectedIds, elements, size])
   131→
   132→  /* ------------------------- keyboard: space pan cursor ------------------------- */
   133→  useEffect(() => {
   134→    function onKeyDown(e: KeyboardEvent) {
   135→      if (e.code === "Space" && !isTextEntry(e.target)) {
   136→        spaceDown.current = true
   137→        if (wrapRef.current && !e.repeat) wrapRef.current.style.cursor = "grab"
   138→      }
   139→    }
   140→    function onKeyUp(e: KeyboardEvent) {
   141→      if (e.code === "Space") {
   142→        spaceDown.current = false
   143→        if (wrapRef.current) wrapRef.current.style.cursor = ""
   144→      }
   145→    }
   146→    window.addEventListener("keydown", onKeyDown)
   147→    window.addEventListener("keyup", onKeyUp)
   148→    return () => {
   149→      window.removeEventListener("keydown", onKeyDown)
   150→      window.removeEventListener("keyup", onKeyUp)
   151→    }
   152→  }, [])
   153→
   154→  /* ------------------------- commit gesture (pan/zoom) ------------------------- */
   155→  const commitGesture = useCallback(() => {
   156→    const layer = contentLayerRef.current
   157→    if (!layer) return
   158→    if (gestureCommit.current) clearTimeout(gestureCommit.current)
   159→    gestureCommit.current = setTimeout(() => {
   160→      api.setView({
   161→        zoom: layer.scaleX(),
   162→        panX: layer.x(),
   163→        panY: layer.y(),
   164→      })
   165→    }, 140)
   166→  }, [api])
   167→
   168→  /* keep layer transform in sync with view state when it changes externally */
   169→  useEffect(() => {
   170→    const layer = contentLayerRef.current
   171→    if (!layer) return
   172→    layer.position({ x: api.view.panX, y: api.view.panY })
   173→    layer.scale({ x: api.view.zoom, y: api.view.zoom })
   174→    layer.batchDraw()
   175→  }, [api.view.zoom, api.view.panX, api.view.panY])
   176→
   177→  /* ------------------------- element handlers ------------------------- */
   178→
   179→  const memberIdsOf = useCallback(
   180→    (el: DesignElement): string[] => {
   181→      if (!el.groupId) return [el.id]
   182→      return elements.filter((m) => m.groupId === el.groupId).map((m) => m.id)
   183→    },
   184→    [elements],
   185→  )
   186→
   187→  const handleSelect = useCallback(
   188→    (el: DesignElement, additive: boolean) => {
   189→      const ids = el.groupId ? memberIdsOf(el) : [el.id]
   190→      api.selectIds(ids, additive)
   191→    },
   192→    [api, memberIdsOf],
   193→  )
   194→
   195→  function onDragStart(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   196→    const layer = contentLayerRef.current
   197→    if (!layer) return
   198→    const orig = new Map<string, { x: number; y: number }>()
   199→    for (const id of memberIdsOf(el)) {
   200→      const node = layer.findOne(`#${id}`)
   201→      if (node) orig.set(id, { x: node.x(), y: node.y() })
   202→    }
   203→    dragState.current = { id: el.id, groupId: el.groupId, orig }
   204→  }
   205→
   206→  function onDragMove(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   207→    const node = e.target
   208→    const layer = contentLayerRef.current
   209→    if (!layer) return
   210→    const zoom = layer.scaleX() || 1
   211→    const selectedSet = new Set(memberIdsOf(el))
   212→    const others = elements
   213→      .filter((o) => !o.hidden && !selectedSet.has(o.id))
   214→      .map((o) => ({ id: o.id, x: o.x, y: o.y, width: o.width, height: o.height }))
   215→    const snap = computeSnap(
   216→      { id: el.id, x: node.x(), y: node.y(), width: el.width, height: el.height },
   217→      others,
   218→      {
   219→        threshold: 6 / zoom,
   220→        gridSize: api.view.snapToGrid ? api.view.gridSize : null,
   221→        pageW: doc.width,
   222→        pageH: doc.height,
   223→      },
   224→    )
   225→    node.position({ x: node.x() + snap.dx, y: node.y() + snap.dy })
   226→    drawGuides(snap.guides)
   227→
   228→    const st = dragState.current
   229→    if (st && el.groupId) {
   230→      const orig = st.orig.get(el.id)
   231→      if (orig) {
   232→        const dx = node.x() - orig.x
   233→        const dy = node.y() - orig.y
   234→        for (const [id, o] of st.orig) {
   235→          if (id === el.id) continue
   236→          const n = layer.findOne(`#${id}`)
   237→          if (n) n.position({ x: o.x + dx, y: o.y + dy })
   238→        }
   239→      }
   240→    }
   241→    layer.batchDraw()
   242→  }
   243→
   244→  function onDragEnd(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   245→    const node = e.target
   246→    const patches: ElementPatch[] = []
   247→    const st = dragState.current
   248→    if (st && el.groupId) {
   249→      const orig = st.orig.get(el.id)
   250→      if (orig) {
   251→        const dx = node.x() - orig.x
   252→        const dy = node.y() - orig.y
   253→        for (const [id, o] of st.orig) {
   254→          if (id === el.id) patches.push({ id, patch: { x: Math.round(node.x()), y: Math.round(node.y()) } })
   255→          else patches.push({ id, patch: { x: Math.round(o.x + dx), y: Math.round(o.y + dy) } })
   256→        }
   257→      }
   258→    } else {
   259→      patches.push({ id: el.id, patch: { x: Math.round(node.x()), y: Math.round(node.y()) } })
   260→    }
   261→    dragState.current = null
   262→    clearGuides()
   263→    api.updateElements(patches)
   264→  }
   265→
   266→  function onTransformEnd(e: Konva.KonvaEventObject<Event>, el: DesignElement) {
   267→    const node = e.target
   268→    const sx = node.scaleX()
   269→    const sy = node.scaleY()
   270→    node.scaleX(1)
   271→    node.scaleY(1)
   272→    let width = Math.max(8, el.width * sx)
   273→    let height = Math.max(8, el.height * sy)
   274→    const patch: Record<string, unknown> = {
   275→      x: Math.round(node.x()),
   276→      y: Math.round(node.y()),
   277→      width: Math.round(width),
   278→      height: Math.round(height),
   279→      rotation: Math.round(node.rotation() * 10) / 10,
   280→    }
   281→    if (el.type === "text") {
   282→      // keep font size constant while resizing the box; grow stored height if needed
   283→      const t = el as TextElement
   284→      const measured = measureTextBlockHeight(
   285→        {
   286→          text: t.text,
   287→          fontFamily: t.fontFamily,
   288→          fontSize: t.fontSize,
   289→          fontWeight: t.fontWeight,
   290→          italic: t.italic,
   291→          uppercase: t.uppercase,
   292→          lineHeight: t.lineHeight,
   293→          letterSpacing: t.letterSpacing,
   294→          listStyle: t.listStyle,
   295→        },
   296→        patch.width as number,
   297→      )
   298→      patch.height = Math.max(measured, Math.round(height))
   299→    }
   300→    api.updateElements([{ id: el.id, patch: patch as Partial<DesignElement> }])
   301→  }
   302→
   303→  /* ------------------------- stage interactions ------------------------- */
   304→
   305→  const getStagePos = useCallback((): { x: number; y: number } | null => {
   306→    const stage = stageRef.current
   307→    if (!stage) return null
   308→    const p = stage.getPointerPosition()
   309→    return p ? { x: p.x, y: p.y } : null
   310→  }, [])
   311→
   312→  function beginPan(x: number, y: number) {
   313→    panState.current = { active: true, lastX: x, lastY: y }
   314→    if (wrapRef.current) wrapRef.current.style.cursor = "grabbing"
   315→  }
   316→
   317→  function movePan(x: number, y: number) {
   318→    const layer = contentLayerRef.current
   319→    const pan = panState.current
   320→    if (!layer || !pan?.active) return
   321→    layer.position({ x: layer.x() + (x - pan.lastX), y: layer.y() + (y - pan.lastY) })
   322→    layer.batchDraw()
   323→    pan.lastX = x
   324→    pan.lastY = y
   325→  }
   326→
   327→  function endPan() {
   328→    if (!panState.current?.active) return
   329→    panState.current = null
   330→    if (wrapRef.current) wrapRef.current.style.cursor = spaceDown.current ? "grab" : ""
   331→    commitGesture()
   332→  }
   333→
   334→  function onStageMouseDown(e: Konva.KonvaEventObject<MouseEvent>) {
   335→    const stage = stageRef.current
   336→    if (!stage) return
   337→    const pos = getStagePos()
   338→    if (!pos) return
   339→    const middleButton = e.evt.button === 1
   340→    if (spaceDown.current || middleButton) {
   341→      e.evt.preventDefault()
   342→      beginPan(pos.x, pos.y)
   343→      return
   344→    }
   345→    if (e.target === stage && canEdit) {
   346→      // begin rubber-band selection
   347→      const layer = contentLayerRef.current
   348→      const rel = layer ? layer.getRelativePointerPosition() : null
   349→      marquee.current = { active: true, startX: rel?.x ?? 0, startY: rel?.y ?? 0, additive: e.evt.shiftKey }
   350→      const m = marqueeRef.current
   351→      if (m) {
   352→        m.position({ x: rel?.x ?? 0, y: rel?.y ?? 0 })
   353→        m.size({ width: 0, height: 0 })
   354→        m.visible(true)
   355→        m.getLayer()?.batchDraw()
   356→      }
   357→    } else if (e.target === stage) {
   358→      api.selectIds([])
   359→    }
   360→  }
   361→
   362→  function onStageMouseMove(e: Konva.KonvaEventObject<MouseEvent>) {
   363→    const pos = getStagePos()
   364→    if (!pos) return
   365→    if (panState.current?.active) {
   366→      movePan(pos.x, pos.y)
   367→      return
   368→    }
   369→    const mq = marquee.current
   370→    const m = marqueeRef.current
   371→    if (mq?.active && m) {
   372→      const layer = contentLayerRef.current
   373→      const rel = layer ? layer.getRelativePointerPosition() : null
   374→      if (!rel) return
   375→      m.position({ x: Math.min(mq.startX, rel.x), y: Math.min(mq.startY, rel.y) })
   376→      m.size({ width: Math.abs(rel.x - mq.startX), height: Math.abs(rel.y - mq.startY) })
   377→      m.getLayer()?.batchDraw()
   378→    }
   379→    void e
   380→  }
   381→
   382→  function onStageMouseUp() {
   383→    if (panState.current?.active) {
   384→      endPan()
   385→      return
   386→    }
   387→    const mq = marquee.current
   388→    const m = marqueeRef.current
   389→    if (mq?.active && m) {
   390→      m.visible(false)
   391→      m.getLayer()?.batchDraw()
   392→      const box = {
   393→        x: Math.min(mq.startX, m.x()),
   394→        y: Math.min(mq.startY, m.y()),
   395→        width: m.width(),
   396→        height: m.height(),
   397→      }
   398→      marquee.current = null
   399→      if (box.width > 3 || box.height > 3) {
   400→        const hits = elements
   401→          .filter((el) => !el.hidden && !el.locked)
   402→          .filter((el) => el.x < box.x + box.width && el.x + el.width > box.x && el.y < box.y + box.height && el.y + el.height > box.y)
   403→          .flatMap((el) => (el.groupId ? memberIdsOf(el) : [el.id]))
   404→        api.selectIds(Array.from(new Set(hits)), mq.additive)
   405→      } else if (!mq.additive) {
   406→        // simple click on empty canvas clears the selection
   407→        api.selectIds([])
   408→      }
   409→    }
   410→  }
   411→
   412→  /* touch: pinch zoom + pan, long-press context menu */
   413→  const touchGesture = useRef<{ startDist: number; startScale: number; startCenter: { x: number; y: number }; startLayerPos: { x: number; y: number } } | null>(null)
   414→
   415→  function onStageTouchStart(e: Konva.KonvaEventObject<TouchEvent>) {
   416→    const stage = stageRef.current
   417→    if (!stage) return
   418→    const touches = e.evt.touches
   419→    const pos = getStagePos()
   420→    // long-press
   421→    if (pos && touches.length === 1) {
   422→      const targetId = e.target === stage ? null : String((e.target as Konva.Node).getAttr("elementId") ?? "")
   423→      longPress.current = {
   424→        timer: setTimeout(() => {
   425→          longPress.current = null
   426→          onContextMenu(pos.x + (stage.container().getBoundingClientRect().left), pos.y + (stage.container().getBoundingClientRect().top), targetId)
   427→          if (navigator.vibrate) navigator.vibrate(10)
   428→        }, 600),
   429→        x: pos.x,
   430→        y: pos.y,
   431→        targetId,
   432→      }
   433→    }
   434→    if (touches.length === 2) {
   435→      if (longPress.current?.timer) clearTimeout(longPress.current.timer)
   436→      const [a, b] = [touches[0], touches[1]]
   437→      const dist = Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY)
   438→      const layer = contentLayerRef.current
   439→      touchGesture.current = {
   440→        startDist: dist || 1,
   441→        startScale: layer?.scaleX() ?? 1,
   442→        startCenter: { x: (a.clientX + b.clientX) / 2, y: (a.clientY + b.clientY) / 2 },
   443→        startLayerPos: { x: layer?.x() ?? 0, y: layer?.y() ?? 0 },
   444→      }
   445→    } else if (e.target === stage) {
   446→      if (pos) beginPan(pos.x, pos.y)
   447→    }
   448→  }
   449→
   450→  function onStageTouchMove(e: Konva.KonvaEventObject<TouchEvent>) {
   451→    const stage = stageRef.current
   452→    const pos = getStagePos()
   453→    if (longPress.current && pos) {
   454→      const moved = Math.hypot(pos.x - longPress.current.x, pos.y - longPress.current.y)
   455→      if (moved > 8 && longPress.current.timer) {
   456→        clearTimeout(longPress.current.timer)
   457→        longPress.current = null
   458→      }
   459→    }
   460→    const touches = e.evt.touches
   461→    const layer = contentLayerRef.current
   462→    if (touches.length === 2 && layer && touchGesture.current) {
   463→      e.evt.preventDefault()
   464→      const [a, b] = [touches[0], touches[1]]
   465→      const dist = Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY)
   466→      const scale = clamp(touchGesture.current.startScale * (dist / touchGesture.current.startDist), MIN_ZOOM, MAX_ZOOM)
   467→      const rect = stage!.container().getBoundingClientRect()
   468→      const cx = touchGesture.current.startCenter.x - rect.left
   469→      const cy = touchGesture.current.startCenter.y - rect.top
   470→      const g = touchGesture.current
   471→      const docX = (cx - g.startLayerPos.x) / g.startScale
   472→      const docY = (cy - g.startLayerPos.y) / g.startScale
   473→      const curCenterX = (a.clientX + b.clientX) / 2 - rect.left
   474→      const curCenterY = (a.clientY + b.clientY) / 2 - rect.top
   475→      layer.scale({ x: scale, y: scale })
   476→      layer.position({ x: curCenterX - docX * scale, y: curCenterY - docY * scale })
   477→      layer.batchDraw()
   478→      commitGesture()
   479→      return
   480→    }
   481→    if (pos && panState.current?.active) movePan(pos.x, pos.y)
   482→  }
   483→
   484→  function onStageTouchEnd() {
   485→    if (longPress.current?.timer) {
   486→      clearTimeout(longPress.current.timer)
   487→      longPress.current = null
   488→    }
   489→    touchGesture.current = null
   490→    endPan()
   491→  }

   493→  function onWheel(e: Konva.KonvaEventObject<WheelEvent>) {
   494→    const layer = contentLayerRef.current
   495→    const stage = stageRef.current
   496→    if (!layer || !stage) return
   497→    e.evt.preventDefault()
   498→    const pos = stage.getPointerPosition()
   499→    if (!pos) return
   500→    if (e.evt.ctrlKey || e.evt.metaKey) {
   501→      const oldScale = layer.scaleX()
   502→      const newScale = clamp(oldScale * Math.exp(-e.evt.deltaY * 0.0016), MIN_ZOOM, MAX_ZOOM)
   503→      const docX = (pos.x - layer.x()) / oldScale
   504→      const docY = (pos.y - layer.y()) / oldScale
   505→      layer.scale({ x: newScale, y: newScale })
   506→      layer.position({ x: pos.x - docX * newScale, y: pos.y - docY * newScale })
   507→      layer.batchDraw()
   508→      commitGesture()
   509→    } else {
   510→      layer.position({ x: layer.x() - e.evt.deltaX, y: layer.y() - e.evt.deltaY })
   511→      layer.batchDraw()
   512→      commitGesture()
   513→    }
   514→  }

   516→  function onStageContextMenu(e: Konva.KonvaEventObject<PointerEvent>) {
   517→    e.evt.preventDefault()
   518→    const stage = stageRef.current
   519→    if (!stage) return
   520→    const targetId = e.target === stage ? null : String((e.target as Konva.Node).getAttr("elementId") ?? "")

   950→function isTextEntry(target: EventTarget | null): boolean {

   956→function gradientEnd(angleDeg: number, w: number, h: number): { x: number; y: number } {
   957→  const a = (angleDeg * Math.PI) / 180
   958→  const dx = Math.cos(a)
   959→  const dy = Math.sin(a)
   960→  const len = Math.abs(dx) * w + Math.abs(dy) * h
   961→  return { x: w / 2 + (dx * len) / 2, y: h / 2 + (dy * len) / 2 }
   962→}

   964→/** Icon path data (cached) from the shared ICON_LIBRARY. */
   965→const iconPathCache = new Map<string, string[] | null>()
   966→function iconPaths(name: string): string[] | null {
   967→  if (iconPathCache.has(name)) return iconPathCache.get(name) ?? null
   968→  const def = ICON_LIBRARY[name]
   969→  iconPathCache.set(name, def ? def.paths : null)
   970→  return def ? def.paths : null
   971→}

   973→function BgImage({ info, width, height }: { info: { source: HTMLCanvasElement | HTMLImageElement; natW: number; natH: number }; width: number; height: number }) {
   974→  if (!info.natW || !info.natH) return null
   975→  const ratio = Math.max(width / info.natW, height / info.natH)
   976→  const dw = info.natW * ratio
   977→  const dh = info.natH * ratio
   978→  return <KImage image={info.source} x={(width - dw) / 2} y={(height - dh) / 2} width={dw} height={dh} listening={false} />
   979→}

   981→function GridShape({ width, height, gridSize }: { width: number; height: number; gridSize: number }) {
   982→  return (
   983→    <KShape
   984→      sceneFunc={(ctx, shape) => {
   985→        ctx.beginPath()
   986→        for (let x = gridSize; x < width; x += gridSize) {
   987→          ctx.moveTo(x, 0)
   988→          ctx.lineTo(x, height)
   989→        }
   990→        for (let y = gridSize; y < height; y += gridSize) {
   991→          ctx.moveTo(0, y)
   992→          ctx.lineTo(width, y)
   993→        }
   994→        ctx.setAttr("strokeStyle", "rgba(100,100,130,0.18)")
   995→        ctx.setAttr("lineWidth", 1)
   996→        ctx.stroke()
   997→      }}
   998→      listening={false}
   999→    />
  1000→  )

  1007→      <Rect width={el.width} height={el.height} fill="#ffffff" cornerRadius={8} listening={false} />
  1008→      {items.map((p, i) => {
  1009→        if (p.kind === "rect") {
  1010→          return <Rect key={i} x={p.x} y={p.y} width={p.width} height={p.height} fill={p.fill} cornerRadius={p.cornerRadius} listening={false} />
  1011→        }
  1012→        if (p.kind === "path") {
  1013→          return <KPath key={i} data={p.d} fill={p.fill} stroke={p.stroke} strokeWidth={p.strokeWidth} listening={false} />
  1014→        }
  1015→        if (p.kind === "polyline") {
  1016→          return <KLine key={i} points={p.points} stroke={p.stroke} strokeWidth={p.strokeWidth} lineJoin="round" lineCap="round" listening={false} />
  1017→        }
  1018→        if (p.kind === "text") {
  1019→          return (
  1020→            <KText
  1021→              key={i}
  1022→              text={p.text}
  1023→              x={p.align === "center" ? p.x - 300 : p.align === "right" ? p.x - 600 : p.x}
  1024→              y={p.y - p.size}
  1025→              width={600}
  1026→              align={p.align}
  1027→              fontSize={p.size}
  1028→              fontFamily="Inter"
  1029→              fontStyle={p.weight && p.weight >= 600 ? "bold" : "normal"}
  1030→              fill={p.color}
  1031→              listening={false}
  1032→            />
  1033→          )
  1034→        }
  1035→        return <KLine key={i} points={[p.x1, p.y, p.x2, p.y]} stroke={p.stroke} strokeWidth={1} dash={p.dash} listening={false} />
  1036→      })}
  1037→    </Group>
  1038→  )
