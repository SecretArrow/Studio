     4→     4→ * StageView — the interactive Konva canvas.
     5→     5→ *
     6→     6→ *  • pan (space+drag, middle-drag, wheel, two-finger touch) & zoom (ctrl+wheel,
     7→     7→ *    pinch, buttons via api.setView)

   106→   106→        strokeWidth: 1 / (viewRef.current.zoom || 1),
   107→   107→        dash: [4 / (viewRef.current.zoom || 1), 4 / (viewRef.current.zoom || 1)],
   108→   108→        listening: false,
   109→   109→      })
   110→   110→      layer.add(line)
   111→   111→    }
   112→   112→    layer.batchDraw()
   113→   113→  }, [])
   114→   114→
   115→   115→  const clearGuides = useCallback(() => {
   116→   116→    const layer = overlayLayerRef.current
   117→   117→    if (!layer) return
   118→   118→    layer.destroyChildren()
   119→   119→    layer.batchDraw()
   120→   120→  }, [])
   121→   121→
   122→   122→  /* ------------------------- transformer sync ------------------------- */
   123→   123→  useEffect(() => {
   124→   124→    const layer = contentLayerRef.current
   125→   125→    const tr = trRef.current
   126→   126→    if (!layer || !tr) return

   128→   128→    tr.nodes(nodes as Konva.Node[])
   129→   129→    tr.getLayer()?.batchDraw()

   131→   131→
   132→   132→  /* ------------------------- keyboard: space pan cursor ------------------------- */
   133→   133→  useEffect(() => {
   134→   134→    function onKeyDown(e: KeyboardEvent) {
   135→   135→      if (e.code === "Space" && !isTextEntry(e.target)) {
   136→   136→        spaceDown.current = true

   138→   138→      }
   139→   139→    }
   140→   140→    function onKeyUp(e: KeyboardEvent) {
   141→   141→      if (e.code === "Space") {
   142→   142→        spaceDown.current = false
   143→   143→        if (wrapRef.current) wrapRef.current.style.cursor = ""
   144→   144→      }
   145→   145→    }
   146→   146→    window.addEventListener("keydown", onKeyDown)
   147→   147→    window.addEventListener("keyup", onKeyUp)
   148→   148→    return () => {
   149→   149→      window.removeEventListener("keydown", onKeyDown)
   150→   150→      window.removeEventListener("keyup", onKeyUp)
   151→   151→    }
   152→   152→  }, [])
   153→   153→
   154→   154→  /* ------------------------- commit gesture (pan/zoom) ------------------------- */
   155→   155→  const commitGesture = useCallback(() => {
   156→   156→    const layer = contentLayerRef.current
   157→   157→    if (!layer) return
   158→   158→    if (gestureCommit.current) clearTimeout(gestureCommit.current)
   159→   159→    gestureCommit.current = setTimeout(() => {
   160→   160→      api.setView({
   161→   161→        zoom: layer.scaleX(),
   162→   162→        panX: layer.x(),
   163→   163→        panY: layer.y(),
   164→   164→      })
   165→   165→    }, 140)
   166→   166→  }, [api])
   167→   167→

   170→   170→    const layer = contentLayerRef.current
   171→   171→    if (!layer) return
   172→   172→    layer.position({ x: api.view.panX, y: api.view.panY })
   173→   173→    layer.scale({ x: api.view.zoom, y: api.view.zoom })
   174→   174→    layer.batchDraw()
   175→   175→  }, [api.view.zoom, api.view.panX, api.view.panY])
   176→   176→
   177→   177→  /* ------------------------- element handlers ------------------------- */
   178→   178→
   179→   179→  const memberIdsOf = useCallback(
   180→   180→    (el: DesignElement): string[] => {
   181→   181→      if (!el.groupId) return [el.id]
   182→   182→      return elements.filter((m) => m.groupId === el.groupId).map((m) => m.id)
   183→   183→    },
   184→   184→    [elements],
   185→   185→  )
   186→   186→

   189→   189→      const ids = el.groupId ? memberIdsOf(el) : [el.id]

   195→   195→  function onDragStart(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   196→   196→    const layer = contentLayerRef.current
   197→   197→    if (!layer) return
   198→   198→    const orig = new Map<string, { x: number; y: number }>()
   199→   199→    for (const id of memberIdsOf(el)) {
   200→   200→      const node = layer.findOne(`#${id}`)
   201→   201→      if (node) orig.set(id, { x: node.x(), y: node.y() })
   202→   202→    }
   203→   203→    dragState.current = { id: el.id, groupId: el.groupId, orig }
   204→   204→  }
   205→   205→
   206→   206→  function onDragMove(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   207→   207→    const node = e.target
   208→   208→    const layer = contentLayerRef.current
   209→   209→    if (!layer) return
   210→   210→    const zoom = layer.scaleX() || 1
   211→   211→    const selectedSet = new Set(memberIdsOf(el))
   212→   212→    const others = elements
   213→   213→      .filter((o) => !o.hidden && !selectedSet.has(o.id))
   214→   214→      .map((o) => ({ id: o.id, x: o.x, y: o.y, width: o.width, height: o.height }))
   215→   215→    const snap = computeSnap(
   216→   216→      { id: el.id, x: node.x(), y: node.y(), width: el.width, height: el.height },
   217→   217→      others,
   218→   218→      {
   219→   219→        threshold: 6 / zoom,
   220→   220→        gridSize: api.view.snapToGrid ? api.view.gridSize : null,
   221→   221→        pageW: doc.width,
   222→   222→        pageH: doc.height,
   223→   223→      },
   224→   224→    )
   225→   225→    node.position({ x: node.x() + snap.dx, y: node.y() + snap.dy })
   226→   226→    drawGuides(snap.guides)
   227→   227→
   228→   228→    const st = dragState.current
   229→   229→    if (st && el.groupId) {
   230→   230→      const orig = st.orig.get(el.id)
   231→   231→      if (orig) {
   232→   232→        const dx = node.x() - orig.x
   233→   233→        const dy = node.y() - orig.y
   234→   234→        for (const [id, o] of st.orig) {
   235→   235→          if (id === el.id) continue
   236→   236→          const n = layer.findOne(`#${id}`)
   237→   237→          if (n) n.position({ x: o.x + dx, y: o.y + dy })
   238→   238→        }
   239→   239→      }
   240→   240→    }
   241→   241→    layer.batchDraw()
   242→   242→  }
   243→   243→
   244→   244→  function onDragEnd(e: Konva.KonvaEventObject<DragEvent>, el: DesignElement) {
   245→   245→    const node = e.target
   246→   246→    const patches: ElementPatch[] = []
   247→   247→    const st = dragState.current
   248→   248→    if (st && el.groupId) {
   249→   249→      const orig = st.orig.get(el.id)
   250→   250→      if (orig) {
   251→   251→        const dx = node.x() - orig.x
   252→   252→        const dy = node.y() - orig.y
   253→   253→        for (const [id, o] of st.orig) {
   254→   254→          if (id === el.id) patches.push({ id, patch: { x: Math.round(node.x()), y: Math.round(node.y()) } })
   255→   255→          else patches.push({ id, patch: { x: Math.round(o.x + dx), y: Math.round(o.y + dy) } })
   256→   256→        }
   257→   257→      }
   258→   258→    } else {
   259→   259→      patches.push({ id: el.id, patch: { x: Math.round(node.x()), y: Math.round(node.y()) } })
   260→   260→    }
   261→   261→    dragState.current = null
   262→   262→    clearGuides()
   263→   263→    api.updateElements(patches)
   264→   264→  }
   265→   265→
   266→   266→  function onTransformEnd(e: Konva.KonvaEventObject<Event>, el: DesignElement) {
   267→   267→    const node = e.target
   268→   268→    const sx = node.scaleX()
   269→   269→    const sy = node.scaleY()
   270→   270→    node.scaleX(1)
   271→   271→    node.scaleY(1)
   272→   272→    let width = Math.max(8, el.width * sx)
   273→   273→    let height = Math.max(8, el.height * sy)
   274→   274→    const patch: Record<string, unknown> = {
   275→   275→      x: Math.round(node.x()),
   276→   276→      y: Math.round(node.y()),
   277→   277→      width: Math.round(width),
   278→   278→      height: Math.round(height),
   279→   279→      rotation: Math.round(node.rotation() * 10) / 10,
   280→   280→    }
   281→   281→    if (el.type === "text") {
   282→   282→      // keep font size constant while resizing the box; grow stored height if needed
   283→   283→      const t = el as TextElement
   284→   284→      const measured = measureTextBlockHeight(
   285→   285→        {
   286→   286→          text: t.text,
   287→   287→          fontFamily: t.fontFamily,
   288→   288→          fontSize: t.fontSize,
   289→   289→          fontWeight: t.fontWeight,
   290→   290→          italic: t.italic,
   291→   291→          uppercase: t.uppercase,
   292→   292→          lineHeight: t.lineHeight,
   293→   293→          letterSpacing: t.letterSpacing,
   294→   294→          listStyle: t.listStyle,
   295→   295→        },
   296→   296→        patch.width as number,
   297→   297→      )
   298→   298→      patch.height = Math.max(measured, Math.round(height))
   299→   299→    }
   300→   300→    api.updateElements([{ id: el.id, patch: patch as Partial<DesignElement> }])
   301→   301→  }

   303→   303→  /* ------------------------- stage interactions ------------------------- */
   304→   304→
   305→   305→  const getStagePos = useCallback((): { x: number; y: number } | null => {
   306→   306→    const stage = stageRef.current
   307→   307→    if (!stage) return null
   308→   308→    const p = stage.getPointerPosition()
   309→   309→    return p ? { x: p.x, y: p.y } : null
   310→   310→  }, [])
   311→   311→
   312→   312→  function beginPan(x: number, y: number) {
   313→   313→    panState.current = { active: true, lastX: x, lastY: y }
   314→   314→    if (wrapRef.current) wrapRef.current.style.cursor = "grabbing"
   315→   315→  }
   316→   316→
   317→   317→  function movePan(x: number, y: number) {
   318→   318→    const layer = contentLayerRef.current
   319→   319→    const pan = panState.current
   320→   320→    if (!layer || !pan?.active) return
   321→   321→    layer.position({ x: layer.x() + (x - pan.lastX), y: layer.y() + (y - pan.lastY) })
   322→   322→    layer.batchDraw()
   323→   323→    pan.lastX = x
   324→   324→    pan.lastY = y
   325→   325→  }
   326→   326→
   327→   327→  function endPan() {
   328→   328→    if (!panState.current?.active) return
   329→   329→    panState.current = null
   330→   330→    if (wrapRef.current) wrapRef.current.style.cursor = spaceDown.current ? "grab" : ""
   331→   331→    commitGesture()
   332→   332→  }
   333→   333→
   334→   334→  function onStageMouseDown(e: Konva.KonvaEventObject<MouseEvent>) {
   335→   335→    const stage = stageRef.current
   336→   336→    if (!stage) return
   337→   337→    const pos = getStagePos()
   338→   338→    if (!pos) return
   339→   339→    const middleButton = e.evt.button === 1
   340→   340→    if (spaceDown.current || middleButton) {
   341→   341→      e.evt.preventDefault()
   342→   342→      beginPan(pos.x, pos.y)
   343→   343→      return
   344→   344→    }
   345→   345→    if (e.target === stage && canEdit) {
   346→   346→      // begin rubber-band selection
   347→   347→      const layer = contentLayerRef.current
   348→   348→      const rel = layer ? layer.getRelativePointerPosition() : null
   349→   349→      marquee.current = { active: true, startX: rel?.x ?? 0, startY: rel?.y ?? 0, additive: e.evt.shiftKey }
   350→   350→      const m = marqueeRef.current
   351→   351→      if (m) {
   352→   352→        m.position({ x: rel?.x ?? 0, y: rel?.y ?? 0 })
   353→   353→        m.size({ width: 0, height: 0 })
   354→   354→        m.visible(true)
   355→   355→        m.getLayer()?.batchDraw()
   356→   356→      }
   357→   357→    } else if (e.target === stage) {
   358→   358→      api.selectIds([])
   359→   359→    }
   360→   360→  }
   361→   361→
   362→   362→  function onStageMouseMove(e: Konva.KonvaEventObject<MouseEvent>) {
   363→   363→    const pos = getStagePos()
   364→   364→    if (!pos) return
   365→   365→    if (panState.current?.active) {
   366→   366→      movePan(pos.x, pos.y)
   367→   367→      return
   368→   368→    }
   369→   369→    const mq = marquee.current
   370→   370→    const m = marqueeRef.current
   371→   371→    if (mq?.active && m) {
   372→   372→      const layer = contentLayerRef.current
   373→   373→      const rel = layer ? layer.getRelativePointerPosition() : null
   374→   374→      if (!rel) return
   375→   375→      m.position({ x: Math.min(mq.startX, rel.x), y: Math.min(mq.startY, rel.y) })
   376→   376→      m.size({ width: Math.abs(rel.x - mq.startX), height: Math.abs(rel.y - mq.startY) })
   377→   377→      m.getLayer()?.batchDraw()
   378→   378→    }
   379→   379→    void e
   380→   380→  }
   381→   381→
   382→   382→  function onStageMouseUp() {
   383→   383→    if (panState.current?.active) {
   384→   384→      endPan()
   385→   385→      return
   386→   386→    }
   387→   387→    const mq = marquee.current
   388→   388→    const m = marqueeRef.current
   389→   389→    if (mq?.active && m) {
   390→   390→      m.visible(false)
   391→   391→      m.getLayer()?.batchDraw()
   392→   392→      const box = {
   393→   393→        x: Math.min(mq.startX, m.x()),
   394→   394→        y: Math.min(mq.startY, m.y()),
   395→   395→        width: m.width(),
   396→   396→        height: m.height(),
   397→   397→      }
   398→   398→      marquee.current = null
   399→   399→      if (box.width > 3 || box.height > 3) {
   400→   400→        const hits = elements
   401→   401→          .filter((el) => !el.hidden && !el.locked)
   402→   402→          .filter((el) => el.x < box.x + box.width && el.x + el.width > box.x && el.y < box.y + box.height && el.y + el.height > box.y)
   403→   403→          .flatMap((el) => (el.groupId ? memberIdsOf(el) : [el.id]))
   404→   404→        api.selectIds(Array.from(new Set(hits)), mq.additive)
   405→   405→      } else if (!mq.additive) {
   406→   406→        // simple click on empty canvas clears the selection
   407→   407→        api.selectIds([])
   408→   408→      }
   409→   409→    }
   410→   410→  }

   412→   412→  /* touch: pinch zoom + pan, long-press context menu */
   413→   413→  const touchGesture = useRef<{ startDist: number; startScale: number; startCenter: { x: number; y: number }; startLayerPos: { x: number; y: number } } | null>(null)
   414→   414→
   415→   415→  function onStageTouchStart(e: Konva.KonvaEventObject<TouchEvent>) {
   416→   416→    const stage = stageRef.current
   417→   417→    if (!stage) return
   418→   418→    const touches = e.evt.touches
   419→   419→    const pos = getStagePos()
   420→   420→    // long-press
   421→   421→    if (pos && touches.length === 1) {
   422→   422→      const targetId = e.target === stage ? null : String((e.target as Konva.Node).getAttr("elementId") ?? "")
   423→   423→      longPress.current = {
   424→   424→        timer: setTimeout(() => {
   425→   425→          longPress.current = null
   426→   426→          onContextMenu(pos.x + (stage.container().getBoundingClientRect().left), pos.y + (stage.container().getBoundingClientRect().top), targetId)
   427→   427→          if (navigator.vibrate) navigator.vibrate(10)
   428→   428→        }, 600),
   429→   429→        x: pos.x,
   430→   430→        y: pos.y,
   431→   431→        targetId,
   432→   432→      }
   433→   433→    }
   434→   434→    if (touches.length === 2) {
   435→   435→      if (longPress.current?.timer) clearTimeout(longPress.current.timer)
   436→   436→      const [a, b] = [touches[0], touches[1]]
   437→   437→      const dist = Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY)
   438→   438→      const layer = contentLayerRef.current
   439→   439→      touchGesture.current = {
   440→   440→        startDist: dist || 1,
   441→   441→        startScale: layer?.scaleX() ?? 1,
   442→   442→        startCenter: { x: (a.clientX + b.clientX) / 2, y: (a.clientY + b.clientY) / 2 },
   443→   443→        startLayerPos: { x: layer?.x() ?? 0, y: layer?.y() ?? 0 },
   444→   444→      }
   445→   445→    } else if (e.target === stage) {
   446→   446→      if (pos) beginPan(pos.x, pos.y)
   447→   447→    }
   448→   448→  }

   450→   450→  function onStageTouchMove(e: Konva.KonvaEventObject<TouchEvent>) {
   451→   451→    const stage = stageRef.current
   452→   452→    const pos = getStagePos()
   453→   453→    if (longPress.current && pos) {
   454→   454→      const moved = Math.hypot(pos.x - longPress.current.x, pos.y - longPress.current.y)
   455→   455→      if (moved > 8 && longPress.current.timer) {
   456→   456→        clearTimeout(longPress.current.timer)
   457→   457→        longPress.current = null
   458→   458→      }
   459→   459→    }
   460→   460→    const touches = e.evt.touches
   461→   461→    const layer = contentLayerRef.current
   462→   462→    if (touches.length === 2 && layer && touchGesture.current) {
   463→   463→      e.evt.preventDefault()
   464→   464→      const [a, b] = [touches[0], touches[1]]
   465→   465→      const dist = Math.hypot(a.clientX - b.clientX, a.clientY - b.clientY)
   466→   466→      const scale = clamp(touchGesture.current.startScale * (dist / touchGesture.current.startDist), MIN_ZOOM, MAX_ZOOM)
   467→   467→      const rect = stage!.container().getBoundingClientRect()
   468→   468→      const cx = touchGesture.current.startCenter.x - rect.left
   469→   469→      const cy = touchGesture.current.startCenter.y - rect.top
   470→   470→      const g = touchGesture.current
   471→   471→      const docX = (cx - g.startLayerPos.x) / g.startScale
   472→   472→      const docY = (cy - g.startLayerPos.y) / g.startScale
   473→   473→      const curCenterX = (a.clientX + b.clientX) / 2 - rect.left
   474→   474→      const curCenterY = (a.clientY + b.clientY) / 2 - rect.top
   475→   475→      layer.scale({ x: scale, y: scale })
   476→   476→      layer.position({ x: curCenterX - docX * scale, y: curCenterY - docY * scale })
   477→   477→      layer.batchDraw()
   478→   478→      commitGesture()
   479→   479→      return
   480→   480→    }
   481→   481→    if (pos && panState.current?.active) movePan(pos.x, pos.y)
   482→   482→  }

   486→   486→      clearTimeout(longPress.current.timer)
   487→   487→      longPress.current = null
   488→   488→    }
   489→   489→    touchGesture.current = null
   490→   490→    endPan()

   493→   493→  function onWheel(e: Konva.KonvaEventObject<WheelEvent>) {
   494→   494→    const layer = contentLayerRef.current
   495→   495→    const stage = stageRef.current
   496→   496→    if (!layer || !stage) return
   497→   497→    e.evt.preventDefault()
   498→   498→    const pos = stage.getPointerPosition()
   499→   499→    if (!pos) return
   500→   500→    if (e.evt.ctrlKey || e.evt.metaKey) {
   501→   501→      const oldScale = layer.scaleX()
   502→   502→      const newScale = clamp(oldScale * Math.exp(-e.evt.deltaY * 0.0016), MIN_ZOOM, MAX_ZOOM)
   503→   503→      const docX = (pos.x - layer.x()) / oldScale
   504→   504→      const docY = (pos.y - layer.y()) / oldScale
   505→   505→      layer.scale({ x: newScale, y: newScale })
   506→   506→      layer.position({ x: pos.x - docX * newScale, y: pos.y - docY * newScale })
   507→   507→      layer.batchDraw()
   508→   508→      commitGesture()
   509→   509→    } else {
   510→   510→      layer.position({ x: layer.x() - e.evt.deltaX, y: layer.y() - e.evt.deltaY })
   511→   511→      layer.batchDraw()
   512→   512→      commitGesture()
   513→   513→    }
   514→   514→  }

   956→   956→function gradientEnd(angleDeg: number, w: number, h: number): { x: number; y: number } {
   957→   957→  const a = (angleDeg * Math.PI) / 180
   958→   958→  const dx = Math.cos(a)
   959→   959→  const dy = Math.sin(a)
   960→   960→  const len = Math.abs(dx) * w + Math.abs(dy) * h
   961→   961→  return { x: w / 2 + (dx * len) / 2, y: h / 2 + (dy * len) / 2 }
   962→   962→}

   973→   973→function BgImage({ info, width, height }: { info: { source: HTMLCanvasElement | HTMLImageElement; natW: number; natH: number }; width: number; height: number }) {
   974→   974→  if (!info.natW || !info.natH) return null
   975→   975→  const ratio = Math.max(width / info.natW, height / info.natH)
   976→   976→  const dw = info.natW * ratio

   978→   978→  return <KImage image={info.source} x={(width - dw) / 2} y={(height - dh) / 2} width={dw} height={dh} listening={false} />
