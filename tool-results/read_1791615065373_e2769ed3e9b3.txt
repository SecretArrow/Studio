    58→function maxEndOnTrack(clips: VideoClip[], trackId: string): number {
    59→  return clips.reduce((max, c) => (c.trackId === trackId ? Math.max(max, clipEnd(c)) : max), 0)

   512→      const t = el as HTMLElement | null
   513→      return !!t && (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable)
   514→    }

   520→      }
   521→      if (isTextEntry(e.target)) return

   527→      } else if (meta && e.key.toLowerCase() === "y") {

   533→      }
   534→    }
   535→    window.addEventListener("keydown", onKey)
   536→    return () => window.removeEventListener("keydown", onKey)

   545→        const d = docRef.current
   546→        if (req.format === "json") {
   547→          return [{ filename: `${base}.studio.json`, blob: new Blob([JSON.stringify(d, null, 2)], { type: "application/json" }) }]
   548→        }
   549→        if (req.format === "png") {
   550→          const cfg = ensureVideoConfig(d)
   551→          const blob = await exportCurrentFramePng(cfg, pool, playheadRef.current)
   552→          return [{ filename: `${base}.png`, blob, note: "Current frame at full canvas size." }]
   553→        }
   554→        if (req.format === "webm" || req.format === "mp4") {
   555→          if (req.format === "mp4" && !(typeof MediaRecorder !== "undefined" && MediaRecorder.isTypeSupported("video/mp4"))) {
   556→            throw new Error("MP4 recording isn't supported by this browser. WebM export is available in the same menu and plays in Chrome, Edge and Firefox.")
   557→          }
   558→          const mime = pickVideoMime(req.format)
   559→          if (!mime) throw new Error("This browser can't record video (MediaRecorder unavailable). Try Chrome, Edge or Firefox.")
   560→          exportingRef.current = true
   561→          setPlaying(false)
   562→          seek(0)
   563→          pool.pauseAll()
   564→          cancelExportRef.current = false
   565→          setExporting({ active: true, progress: 0 })
   566→          try {
   567→            const cfg = ensureVideoConfig(d)
   568→            const { blob, note } = await recordTimeline({
   569→              config: cfg,
   570→              pool,
   571→              mimeType: mime,
   572→              onProgress: (p) => setExporting({ active: true, progress: p }),
   573→              isCancelled: () => cancelExportRef.current,
   574→            })
   575→            if (!blob) return []
   576→            return [
   577→              {
   578→                filename: `${base}.${req.format}`,
   579→                blob,
   580→                note: note ?? `Realtime render — export took about ${fmtClock(totalDurationMs(cfg))} (one pass over the timeline).`,
   581→              },
   582→            ]
   583→          } finally {
   584→            exportingRef.current = false
   585→            setExporting({ active: false, progress: 0 })
   586→          }
   587→        }
   588→        throw new Error(`"${req.format}" isn't available for video projects — use WebM, MP4, PNG or JSON.`)
   589→      },
   590→      getThumbnail: async () => {
   591→        try {
   592→          return await renderThumbnail(ensureVideoConfig(docRef.current), pool, playheadRef.current)
   593→        } catch {
   594→          return null
   595→        }
   596→      },
   597→      isDirty: () => dirtyRef.current,
   598→    }
   599→    registerHandle(handle)
   600→    return () => registerHandle(null)
   601→  }, [registerHandle, pool, seek])
   602→
   603→  /* ---------------- render ---------------- */
   604→
   605→  const canvasPresetValue = useMemo(() => {
   606→    const match = ASPECT_PRESETS.find((p) => p.width === config.canvas.width && p.height === config.canvas.height)
   607→    return match ? match.id : "custom"
   608→  }, [config.canvas.width, config.canvas.height])
   609→
   610→  return (
   611→    <div className="flex h-full min-h-0 flex-col bg-muted/30">
   612→      {/* top toolbar */}
   613→      <div className="flex flex-wrap items-center gap-1.5 border-b bg-card px-2 py-1.5">
   614→        <Button size="sm" variant="outline" className="h-8 gap-1.5" disabled={!canEdit || importing} onClick={() => openPicker(null)}>
   615→          {importing ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Upload className="h-3.5 w-3.5" />}
   616→          Import media
   617→        </Button>
   618→        <Button size="sm" variant="outline" className="h-8 gap-1.5" disabled={!canEdit} onClick={() => addTextClip("text")}>
   619→          <TypeIcon className="h-3.5 w-3.5" /> Text
   620→        </Button>
   621→        <div className="flex items-center gap-1.5">
   622→          <MonitorPlay className="h-3.5 w-3.5 text-muted-foreground" />
   623→          <Select value={canvasPresetValue} disabled={!canEdit} onValueChange={applyAspect}>
   624→            <SelectTrigger className="h-8 w-[130px] text-xs" aria-label="Canvas aspect ratio">
   625→              <SelectValue placeholder="Aspect" />
   626→            </SelectTrigger>
   627→            <SelectContent>
   628→              {ASPECT_PRESETS.map((p) => (
   629→                <SelectItem key={p.id} value={p.id}>
   630→                  {p.label} · {p.width}×{p.height}
   631→                </SelectItem>
   632→              ))}
   633→              {canvasPresetValue === "custom" && (
   634→                <SelectItem value="custom">
   635→                  {config.canvas.width}×{config.canvas.height}
   636→                </SelectItem>
   637→              )}
   638→            </SelectContent>
   639→          </Select>
   640→        </div>
   641→        <div className="ml-auto flex items-center gap-1">
   642→          <span className="mr-1 hidden items-center gap-1 text-[11px] text-muted-foreground sm:flex">
   643→            <Clock className="h-3 w-3" /> {fmtClock(total)}
   644→          </span>
   645→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={undo} disabled={!historyState.canUndo} aria-label="Undo" title="Undo (Ctrl+Z)">
   646→            <Undo2 className="h-4 w-4" />
   647→          </Button>
   648→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={redo} disabled={!historyState.canRedo} aria-label="Redo" title="Redo (Ctrl+Y)">
   649→            <Redo2 className="h-4 w-4" />
   650→          </Button>
   651→        </div>
   652→      </div>
   653→
   654→      {/* session media notice */}
   655→      {hasMedia && (
   656→        <div className="flex items-center gap-1.5 border-b bg-amber-500/10 px-3 py-1 text-[11px] text-amber-900 dark:text-amber-200">
   657→          <Info className="h-3 w-3 shrink-0" />
   658→          <span>
   659→            Media files live in this browser session; re-link after reload.
   660→            {deadCount > 0 && ` ${deadCount} clip${deadCount > 1 ? "s" : ""} need${deadCount === 1 ? "s" : ""} relinking.`}
   661→          </span>
   662→        </div>
   663→      )}
   664→
   665→      <input
   666→        ref={fileInputRef}
   667→        type="file"
   668→        multiple
   669→        accept="video/*,audio/*,image/*"
   670→        className="hidden"
   671→        aria-hidden
   672→        onChange={(e) => {
   673→          void handleFiles(e.target.files)
   674→          e.target.value = ""
   675→        }}
   676→      />
   677→
   678→      <div className="flex min-h-0 flex-1 flex-col lg:flex-row">
   679→        {/* preview */}
   680→        <div className="relative flex min-h-[260px] min-w-0 flex-1 flex-col lg:flex-[1.5]">
   681→          <PreviewPlayer
   682→            config={config}
   683→            playing={playing}
   684→            suspended={exporting.active}
   685→            playheadRef={playheadRef}
   686→            playheadMs={playheadMs}
   687→            pool={pool}
   688→            deadCount={deadCount}
   689→            onTogglePlay={togglePlay}
   690→            onStop={stopPlayback}
   691→            onEnded={handleEnded}
   692→            onTimeUpdate={onTimeUpdate}
   693→          />
   694→          {exporting.active && (
   695→            <div className="absolute inset-0 z-30 flex flex-col items-center justify-center gap-3 bg-background/85 p-6 backdrop-blur-sm">
   696→              <Film className="h-5 w-5 animate-pulse text-primary" />
   697→              <p className="text-sm font-semibold">Rendering video in real time…</p>
   698→              <Progress value={Math.round(exporting.progress * 100)} className="w-64" />
   699→              <p className="text-center text-xs text-muted-foreground">
   700→                {Math.round(exporting.progress * 100)}% · keep this tab visible until it finishes
   701→              </p>
   702→              <Button
   703→                variant="outline"
   704→                size="sm"
   705→                onClick={() => {
   706→                  cancelExportRef.current = true
   707→                }}
   708→              >
   709→                Cancel export
   710→              </Button>
   711→            </div>
   712→          )}
   713→        </div>
   714→
   715→        {/* properties */}
   716→        <div className={cn("flex min-h-0 w-full shrink-0 flex-col border-t bg-card lg:w-80 lg:border-l lg:border-t-0")}>
   717→          <PropertiesPanel
   718→            config={config}
   719→            selected={selected}
   720→            canEdit={canEdit}
   721→            playheadMs={playheadMs}
   722→            onPatchClip={patchClip}
   723→            onAddText={() => addTextClip("text")}
   724→            onAddSticker={(char) => addTextClip("sticker", char)}
   725→            onAddSubtitle={addSubtitle}
   726→            onPatchSubtitle={patchSubtitle}
   727→            onDeleteSubtitle={deleteSubtitle}
   728→            onImportSrt={importSrt}
   729→            onDownloadSrt={downloadSrt}
   730→            isDead={(clipId) => deadSet.has(clipId)}
   731→            onRelink={(clipId) => openPicker(clipId)}
   732→            onDeleteClip={deleteClipById}
   733→          />
   734→        </div>
   735→      </div>
   736→
   737→      {/* timeline */}
   738→      <Timeline
   739→        config={config}
   740→        total={total}
   741→        zoom={zoom}
   742→        onZoomChange={setZoom}
   743→        playheadMs={playheadMs}
   744→        onScrub={seek}
   745→        selectedId={selectedId}
   746→        onSelect={setSelectedId}
   747→        canEdit={canEdit}
   748→        onPatchClip={patchClip}
   749→        onPatchTrack={patchTrack}
   750→        onSplit={splitAtPlayhead}
   751→        onDuplicate={duplicateSelected}
   752→        onDelete={deleteSelected}
   753→        deadIds={deadSet}
   754→        onRelink={(clipId) => openPicker(clipId)}
   755→        getSourceDuration={(clipId) => pool.sourceDurations.get(clipId)}
   756→      />
   757→    </div>
   758→  )
   759→})
   760→
   761→export default VideoEditor
